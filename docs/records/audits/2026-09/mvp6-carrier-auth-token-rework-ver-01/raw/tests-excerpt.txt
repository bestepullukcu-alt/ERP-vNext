   141	    private static PlatformServiceTokenValidator CreateValidator(int clockSkewSeconds = 0)
   142	    {
   143	        var options = ValidOptions();
   144	        options.ClockSkewSeconds = clockSkewSeconds;
   145	        return new PlatformServiceTokenValidator(Options.Create(options));
   146	    }
   147	
   148	    private static PlatformServiceIdentityOptions ValidOptions() => new()
   149	    {
   150	        Enabled = true,
   151	        Issuer = "diten-platform-service",
   152	        Audience = "diten-mdm-reference-validation",
   153	        CallerId = "Diten.Platform",
   154	        KeyId = "platform-mdm-2026-09",
   155	        Secret = Secret,
   156	        MaximumTokenLifetimeSeconds = 60,
   157	        ClockSkewSeconds = 0
   158	    };
   159	
   160	    private static string CreateToken(
   161	        string caller = "Diten.Platform",
   162	        string scope = "mdm.legal-entities.reference.validate",
   163	        string secret = Secret,
   164	        string? audience = "diten-mdm-reference-validation",
   165	        string? omitClaim = null,
   166	        IEnumerable<Claim>? extraClaims = null,
   167	        long? iat = null,
   168	        long? nbf = null,
   169	        long? exp = null)
   170	    {
   171	        var now = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
   172	        var issuedAt = iat ?? now;
   173	        var notBefore = nbf ?? issuedAt;
   174	        var expiresAt = exp ?? issuedAt + 30;
   175	        var claims = new List<Claim>();
   176	        Add(JwtRegisteredClaimNames.Sub, caller);
   177	        Add("scope", scope);
   178	        Add("tenant_id", TenantId.ToString("D"));
   179	        Add("actor_id", ActorId.ToString("D"));
   180	        Add("legal_entity_id", LegalEntityId.ToString("D"));
   181	        Add(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString("D"));
   182	        Add(JwtRegisteredClaimNames.Iat, issuedAt.ToString(), ClaimValueTypes.Integer64);
   183	        if (extraClaims is not null) claims.AddRange(extraClaims);
   184	
   185	        var token = new JwtSecurityToken(
   186	            issuer: "diten-platform-service",
   187	            audience: audience,
   188	            claims: claims,
   189	            notBefore: DateTimeOffset.FromUnixTimeSeconds(notBefore).UtcDateTime,
   190	            expires: DateTimeOffset.FromUnixTimeSeconds(expiresAt).UtcDateTime,
   191	            signingCredentials: new SigningCredentials(
   192	                new SymmetricSecurityKey(Encoding.UTF8.GetBytes(secret)),
   193	                SecurityAlgorithms.HmacSha256));
   194	        token.Header["kid"] = "platform-mdm-2026-09";
   195	        return new JwtSecurityTokenHandler().WriteToken(token);
   196	
   197	        void Add(string type, string value, string valueType = ClaimValueTypes.String)
   198	        {
   199	            if (!string.Equals(omitClaim, type, StringComparison.Ordinal))
   200	                claims.Add(new Claim(type, value, valueType));
   201	        }
   202	    }
