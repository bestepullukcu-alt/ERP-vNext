def raw_span(text, key):
    # locate top-level "key": then bracket-match the array, string-aware
    i = text.index('"%s"' % key); i = text.index('[', i)
    depth = 0; ins = False; esc = False
    for j in range(i, len(text)):
        c = text[j]
        if ins:
            if esc: esc = False
            elif c == '\\': esc = True
            elif c == '"': ins = False
        elif c == '"': ins = True
        elif c == '[': depth += 1
        elif c == ']':
            depth -= 1
            if depth == 0: return text[i:j+1]
    raise SystemExit("unbalanced")
