## 3. Runtime probe (`probe.ps1`)

```powershell
# WP-0176-F2 runtime probe — the built InventoryService on ITS OWN port (5167) with ITS OWN mongod (27417) and an isolated database.
# Never touches the owner's fleet: no request to 4010/5000/5001/5056-5061/5066/5067/27017; upstream masters point to a closed port;
# PlatformRegistration is unset. No token is signed: the probe proves /health and the 401 of the posting / KF-D routes; the authorized
# flow is the integration host (real HTTP pipeline). The fleet's listening PIDs are recorded before and after.
$ErrorActionPreference = 'Stop'
$scratch = Split-Path -Parent $MyInvocation.MyCommand.Path
$repo = 'C:\Users\Ali\erp-vnext-wt-0176'
$mongod = 'C:\Program Files\MongoDB\Server\9.0\bin\mongod.exe'
$mongoPort = 27417
$svcPort = 5167
$db = 'inventory_f2_probe_tests'
$fleetPorts = @(4010, 5000, 5001, 5056, 5057, 5058, 5059, 5060, 5061, 5066, 5067, 27017)
$out = Join-Path $scratch 'probe'
New-Item -ItemType Directory -Force $out | Out-Null
$dbPath = Join-Path $out 'mongo-data'
New-Item -ItemType Directory -Force $dbPath | Out-Null
$result = [ordered]@{}

function Fleet {
    $rows = [ordered]@{}
    foreach ($p in $fleetPorts) {
        $c = Get-NetTCPConnection -State Listen -LocalPort $p -ErrorAction SilentlyContinue | Select-Object -First 1
        $rows["$p"] = if ($c) { $c.OwningProcess } else { $null }
    }
    return $rows
}
$result.fleetBefore = Fleet

$m = Start-Process -FilePath $mongod -ArgumentList @('--port', $mongoPort, '--bind_ip', '127.0.0.1', '--dbpath', "`"$dbPath`"", '--logpath', "`"$(Join-Path $out 'mongod.log')`"") -PassThru -WindowStyle Hidden
$result.mongodPid = $m.Id
Start-Sleep -Seconds 3

$secret = -join ((1..64) | ForEach-Object { '{0:x}' -f (Get-Random -Maximum 16) })
$env:ASPNETCORE_ENVIRONMENT = 'Testing'
$env:ASPNETCORE_URLS = "http://127.0.0.1:$svcPort"
$env:Kestrel__Endpoints__Http__Url = "http://127.0.0.1:$svcPort"
$env:Mongo__ConnectionString = "mongodb://127.0.0.1:$mongoPort"
$env:Mongo__DatabaseName = $db
$env:JwtSettings__Secret = $secret
$env:Inventory__Upstream__ProductMasterBaseUrl = 'http://127.0.0.1:9/'
$env:Inventory__Upstream__LocationMasterBaseUrl = 'http://127.0.0.1:9/'
$env:Inventory__Upstream__LegalEntityMasterBaseUrl = 'http://127.0.0.1:9/'
$env:Inventory__Upstream__OrgUnitDirectoryBaseUrl = 'http://127.0.0.1:9/'
$env:Gateway__BaseUrl = 'http://127.0.0.1:9/'
$env:PlatformRegistration__BaseUrl = ''
$dll = Join-Path $repo 'services\Diten.InventoryService\src\Diten.InventoryService.Api\bin\Debug\net8.0\Diten.InventoryService.Api.dll'
$svcLog = Join-Path $out 'service.log'
$s = Start-Process -FilePath 'dotnet' -ArgumentList @("`"$dll`"", '--urls', "http://127.0.0.1:$svcPort") -WorkingDirectory (Split-Path $dll) `
    -RedirectStandardOutput $svcLog -RedirectStandardError (Join-Path $out 'service.err.log') -PassThru -WindowStyle Hidden
$result.servicePid = $s.Id

try {
    $deadline = (Get-Date).AddSeconds(60)
    do {
        Start-Sleep -Seconds 1
        try { $health = Invoke-WebRequest -Uri "http://127.0.0.1:$svcPort/health" -UseBasicParsing -TimeoutSec 3 } catch { $health = $null }
    } until ($health -or (Get-Date) -gt $deadline)
    $result.health = if ($health) { "$($health.StatusCode) $($health.Content)" } else { 'no answer' }

    function Probe($method, $path) {
        try {
            $r = Invoke-WebRequest -Uri "http://127.0.0.1:$svcPort$path" -Method $method -UseBasicParsing -TimeoutSec 5 -ContentType 'application/json' -Body $(if ($method -in 'PUT','POST','PATCH') { '{}' } else { $null })
            return "$($r.StatusCode)"
        } catch { return "$([int]$_.Exception.Response.StatusCode)" }
    }
    $result.unauthenticated = [ordered]@{
        'POST /api/inventory/movements' = Probe 'POST' '/api/inventory/movements'
        'POST /api/expiry/date-overrides' = Probe 'POST' '/api/expiry/date-overrides'
        'POST /api/trace/lots' = Probe 'POST' '/api/trace/lots'
        "PATCH /api/trace/lots/{id}" = Probe 'PATCH' "/api/trace/lots/$([guid]::NewGuid())"
    }
    Start-Sleep -Seconds 2
    $result.startupLog = @(Get-Content $svcLog | Select-String -Pattern 'MOD-0176|Mongo indexes|Now listening|registration_skipped|PlatformRegistration' | ForEach-Object { $_.Line.Trim() })
}
finally {
    Stop-Process -Id $s.Id -Force -ErrorAction SilentlyContinue
    Stop-Process -Id $m.Id -Force -ErrorAction SilentlyContinue
    Start-Sleep -Seconds 1
    $result.stoppedOwnPids = @($s.Id, $m.Id | Where-Object { -not (Get-Process -Id $_ -ErrorAction SilentlyContinue) })
}
$result.fleetAfter = Fleet
$result.fleetUnchanged = (($result.fleetBefore | ConvertTo-Json) -eq ($result.fleetAfter | ConvertTo-Json))
$result | ConvertTo-Json -Depth 5 | Out-File -Encoding utf8 (Join-Path $out 'probe-result.json')
$result | ConvertTo-Json -Depth 5
```
