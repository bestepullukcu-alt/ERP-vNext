## 1. `sabotage.js` — K3 sabotajları (a)–(e)

Kullanım: `node sabotage.js <worktree> <a..e> apply|restore`. Her sabotaj tek bir metin değişikliğidir; çapa bulunamazsa betik
hata verir. `restore` yedeği bayt bayt geri yazar. Koşu: Inventory test projesini derle, ardından
`--filter "FullyQualifiedName~ProductMasterV11RulesTests|…~ItemLevelLedger|…~LotFlagFromSku|…~TestMongoGuard"` (66 test), ardından
`restore` ve `git status --porcelain -- services` (boş olmalı).

```js
// WP-0173-PM1 K3 sabotages: node sabotage.js <worktree> <a..e> apply|restore. One textual change each (fails loudly when the
// anchor is missing); restore writes the backup back byte for byte.
const fs = require('fs');

const [wt, which, mode] = process.argv.slice(2);
const rules = path.join(wt, 'services/Diten.InventoryService/src/Diten.InventoryService.Domain/Rules/ProductMasterStockRules.cs');

  // (d) an unknown status counted as Active
  d: [rules, '        _ => StockLifecycle.NotUsable\n    };', '        _ => StockLifecycle.Active\n    };'],
  // (e) TestSettings falls back to 27017 again
  e: [settings, 'public static string MongoConnectionString => TestMongo.ConnectionString;', 'public static string MongoConnectionString =>\n        Environment.GetEnvironmentVariable("INVENTORY_TEST_MONGO") ?? "mongodb://127.0.0.1:27017";']
```

## 2. `pm-stub.js` — yerel PRODUCT-MASTER v1.1 taklidi (çalışma zamanı yoklaması)

Prism her işlem için tek bir sabit örnek döndürür; bu yüzden farklı durumlarda birden çok kalem tutamaz. Taklit, draft.8'in
şekillerini bellek içi bir veri kümesiyle sunar. `POST /__control` bir kaydın durumunu ya da bayraklarını değiştirir; kaleme
verilen bir durum SKU'larına da yayılır. `GET /__calls` işlem başına çağrı sayısını döndürür. Yalnız 127.0.0.1:14036'yı dinler.

```js
// WP-0173-PM1 probe — a small LOCAL PRODUCT-MASTER v1.1 imitation (the Prism mock answers ONE fixed example per operation, so
// it cannot hold several items in different states). Shapes follow product-master-bundle.openapi.yaml 1.1.0-draft.8. State is
// in memory; POST /__control changes a record (lifecycle / flags) and GET /__calls returns how many calls each operation got.
const http = require('http');
const data = JSON.parse(require('fs').readFileSync(process.argv[2], 'utf8')); // { products: {id: {...}}, skus: {id: {...}} }
const calls = {};
const count = (op) => { calls[op] = (calls[op] || 0) + 1; };
const send = (r, status, body) => { r.statusCode = status; r.setHeader('content-type', 'application/json'); r.end(JSON.stringify(body)); };
http.createServer((q, r) => {
  const u = new URL(q.url, 'http://x');
  let m;
  let body = '';
  q.on('data', (c) => (body += c));
  q.on('end', () => {
    if (q.method === 'POST' && u.pathname === '/__control') {
      const c = JSON.parse(body);
      const target = data.skus[c.id] || data.products[c.id];
      Object.assign(target || {}, c.set);
      if (data.products[c.id] && c.set.lifecycleStatus) Object.values(data.skus).filter((s) => s.itemId === c.id).forEach((s) => (s.lifecycleStatus = c.set.lifecycleStatus));
      return send(r, target ? 200 : 404, { ok: !!target });
    }
    if (u.pathname === '/__calls') return send(r, 200, calls);
    if (q.method === 'POST' && u.pathname === '/validate') {
      count('validateRefs');
      const refs = JSON.parse(body).refs || [];
      const results = refs.map((x) => {
        if (x.itemId && !x.skuId) {
          const p = data.products[x.itemId];
          return p ? { itemId: x.itemId, exists: true, itemKind: p.itemKind, lifecycleStatus: p.lifecycleStatus } : { itemId: x.itemId, exists: false };
        }
        const s = data.skus[x.skuId];
        const ok = s && (!x.skuLevel || s.skuLevel === x.skuLevel);
        return ok ? { skuId: x.skuId, skuLevel: s.skuLevel, exists: true, itemKind: s.itemKind, lifecycleStatus: s.lifecycleStatus } : { skuId: x.skuId, skuLevel: x.skuLevel, exists: false };
      });
      return send(r, 200, { results, contractVersion: 'v1.1' });
    }
    if ((m = u.pathname.match(/^\/products\/([0-9a-f-]{36})$/))) {
      count('getProduct');
      const p = data.products[m[1]];
      return p ? send(r, 200, Object.assign({}, p, { contractVersion: 'v1.1' })) : send(r, 404, { error: { code: 'UNKNOWN_ITEM' } });
    }
    if ((m = u.pathname.match(/^\/skus\/([0-9a-f-]{36})\/uom$/))) {
      count('getSkuUom');
      const s = data.skus[m[1]];
      return s ? send(r, 200, { baseUomId: s.baseUomId, conversions: s.conversions || [], contractVersion: 'v1.1' }) : send(r, 404, { error: { code: 'UNKNOWN_SKU' } });
    }
    if ((m = u.pathname.match(/^\/skus\/([0-9a-f-]{36})$/))) {
      count('getSku');
      const s = data.skus[m[1]];
      const level = u.searchParams.get('level');
      return s && (!level || s.skuLevel === level) ? send(r, 200, Object.assign({}, s, { contractVersion: 'v1.1' })) : send(r, 404, { error: { code: 'UNKNOWN_SKU' } });
    }
    if (u.pathname === '/stockable-items') {
      count('searchStockableItems');
      const status = u.searchParams.get('status');
      const qq = (u.searchParams.get('q') || '').toLowerCase();
      const page = Number(u.searchParams.get('page') || 1);
      const size = Number(u.searchParams.get('pageSize') || 25);
      const rows = Object.values(data.skus)
        .filter((s) => s.stockable !== false)
        .filter((s) => (status ? s.lifecycleStatus === status : s.lifecycleStatus !== 'Retired'))
        .filter((s) => !qq || s.canonicalCode.toLowerCase().startsWith(qq) || s.name.toLowerCase().includes(qq))
        .sort((a, b) => (a.canonicalCode < b.canonicalCode ? -1 : 1))
        .map((s) => ({ skuId: s.skuId, skuLevel: s.skuLevel, itemId: s.itemId, itemKind: s.itemKind, canonicalCode: s.canonicalCode, name: s.name, materialType: s.materialType ?? null, lifecycleStatus: s.lifecycleStatus, baseUomId: s.baseUomId, lotControlled: s.lotControlled, serialControlled: s.serialControlled }));
      return send(r, 200, { items: rows.slice((page - 1) * size, page * size), page, pageSize: size, totalCount: rows.length, contractVersion: 'v1.1' });
    }
    send(r, 404, {});
  });
}).listen(14036, '127.0.0.1');
```
