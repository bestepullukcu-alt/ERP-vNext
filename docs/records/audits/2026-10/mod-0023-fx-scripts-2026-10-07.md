    // ── fixture: test password on three seeded users of the default tenant (probe-owned database only) ──────────────
    var hash = BCrypt.Net.BCrypt.HashPassword(testPassword, 12);
    foreach (var dbName in (await mongo.ListDatabaseNamesAsync()).ToList())
    {
        Log($"diag db {dbName}");
    }
    var userDocs = await users.Find(FilterDefinition<BsonDocument>.Empty).Limit(20).ToListAsync();
    Log($"diag users in fx_probe_auth: {await users.CountDocumentsAsync(FilterDefinition<BsonDocument>.Empty)} → {string.Join(", ", userDocs.Select(u => u.GetValue("Email", "?") + "@" + u.GetValue("TenantId", BsonNull.Value)))}");
    async Task<Guid> PrepareUser(string email)
    {
        var filter = Builders<BsonDocument>.Filter.Eq("Email", email) & Builders<BsonDocument>.Filter.Eq("TenantId", new BsonBinaryData(tenantId, GuidRepresentation.Standard));
        var doc = await users.Find(filter).FirstOrDefaultAsync()
            ?? await users.Find(Builders<BsonDocument>.Filter.Eq("Email", email)).FirstAsync();
        // Filter by e-mail + tenant: a binary-GUID _id filter matches nothing with this driver setup (measured).
        var tenantOfDoc = doc["TenantId"];
        var byEmailAndTenant = Builders<BsonDocument>.Filter.Eq("Email", email); // the .def e-mails are unique in the collection
        var upd = await users.UpdateOneAsync(byEmailAndTenant, Builders<BsonDocument>.Update
            .Set("PasswordHash", hash).Set("IsActive", true).Set("EmailConfirmed", true).Set("MustChangePassword", false)
            .Set("FailedLoginAttempts", 0).Set("LockoutEnd", BsonNull.Value));
        Log($"diag {email}: _id type={doc["_id"].BsonType} subtype={(doc["_id"].IsBsonBinaryData ? doc["_id"].AsBsonBinaryData.SubType.ToString() : "-")} matched={upd.MatchedCount} byIdCount={await users.CountDocumentsAsync(Builders<BsonDocument>.Filter.Eq("_id", doc["_id"]))} byEmailCount={await users.CountDocumentsAsync(Builders<BsonDocument>.Filter.Eq("Email", email))}");
        var after = await users.Find(Builders<BsonDocument>.Filter.Eq("Email", email)).FirstAsync();
        Log($"fixture {email}: matched={upd.MatchedCount} modified={upd.ModifiedCount} verifiesNow={BCrypt.Net.BCrypt.Verify(testPassword, after["PasswordHash"].AsString)} platformActor={after.GetValue("PlatformActorType", BsonNull.Value)}");
        return doc["_id"].AsBsonBinaryData.ToGuid(doc["_id"].AsBsonBinaryData.SubType == BsonBinarySubType.UuidStandard ? GuidRepresentation.Standard : GuidRepresentation.CSharpLegacy);
    }
    // The seeded platform admin row is rewritten by Platform's administrator sync at startup, so the probe's tenant
    // administrator is a seeded tenant user who is given the tenant's seeded SuperAdmin role: the admin's own
    // userRoles row is cloned with only _id and UserId changed (same schema, same GUID encoding as production writes).
    var adminId = await PrepareUser("charlie.brown.def@diten.com");
    var userRoles = authDb.GetCollection<BsonDocument>("userRoles");
    static Guid G(BsonValue v) => v.AsBsonBinaryData.ToGuid(v.AsBsonBinaryData.SubType == BsonBinarySubType.UuidStandard ? GuidRepresentation.Standard : GuidRepresentation.CSharpLegacy);
    var superAdminRole = (await authDb.GetCollection<BsonDocument>("roles").Find(Builders<BsonDocument>.Filter.Eq("Name", "SuperAdmin")).ToListAsync())
        .First(r => G(r["TenantId"]) == tenantId);

    Log($"fixture SuperAdmin role row for tenant admin: UserId subtype={back["UserId"].AsBsonBinaryData.SubType} RoleId subtype={back["RoleId"].AsBsonBinaryData.SubType}");

    var bId = await PrepareUser("jane.smith.def@diten.com");
    Log($"users: tenant-admin(charlie, SuperAdmin role)={adminId} A={aId} B={bId}");

    async Task<(int Status, JsonNode? Body)> Send(HttpMethod method, string url, string? token, object? body)
    {
        using var req = new HttpRequestMessage(method, url);
        req.Headers.Add("X-Tenant-Id", tenantId.ToString());
        if (token is not null) req.Headers.Authorization = new("Bearer", token);
        if (body is not null) req.Content = JsonContent.Create(body);
        using var res = await http.SendAsync(req);
        var text = await res.Content.ReadAsStringAsync();
        JsonNode? node = null;
        try { node = string.IsNullOrWhiteSpace(text) ? null : JsonNode.Parse(text); } catch { node = JsonValue.Create(text); }
        return ((int)res.StatusCode, node);
    }
    static string? Find(JsonNode? n, string prop)
    {
        if (n is JsonObject o)
        {
            foreach (var kv in o)
            {
                if (string.Equals(kv.Key, prop, StringComparison.OrdinalIgnoreCase) && kv.Value is JsonValue) return kv.Value!.ToString();
                var inner = Find(kv.Value, prop); if (inner is not null) return inner;
            }
        }
        if (n is JsonArray a) foreach (var x in a) { var inner = Find(x, prop); if (inner is not null) return inner; }
        return null;
    }
    static string Reason(JsonNode? n) => Find(n, "reason_code") ?? Find(n, "reasonCode") ?? Find(n, "errorCode") ?? (n?["errors"]?.ToJsonString() ?? "");

    async Task<string> Login(string email)
    {
        var (s, b) = await Send(HttpMethod.Post, AuthUrl + "/api/tenant-auth/login", null, new { email, password = testPassword, rememberMe = false });
        var token = Find(b, "accessToken");
        if (s != 200 || token is null)
        {
            var u = await users.Find(Builders<BsonDocument>.Filter.Eq("Email", email)).ToListAsync();
            foreach (var d in u)
            {
                Log($"diag {email}: fields=[{string.Join(",", d.Names)}] tenant={d.GetValue("TenantId", BsonNull.Value)} hashVerifiesLocally={BCrypt.Net.BCrypt.Verify(testPassword, d.GetValue("PasswordHash", "").ToString())}");
            }
            var events = await authDb.GetCollection<BsonDocument>("authAuditLogs").Find(FilterDefinition<BsonDocument>.Empty)
                .Sort(Builders<BsonDocument>.Sort.Descending("OccurredAt")).Limit(5).ToListAsync();
            Log("diag last audit events: " + string.Join(", ", events.Select(e => e.GetValue("EventName", "?").ToString())));
            throw new InvalidOperationException($"login {email}: {s} {b?.ToJsonString()}");
        }
        return token;
    }

    Check("admin publishes template", ps is 200 or 201, $"{ps} {Reason(pb)}");

    async Task<string> StartAs(string token, string objectId, Guid candidate)
    {
        var (s, b) = await Send(HttpMethod.Post, PlatformUrl + "/api/v1/workflow/instances", token, new
        {
            templateCode = code, objectType = "fx.probe", objectId, candidatePrincipalIds = new[] { "user:" + candidate },
            reasonCode = "SUBMITTED", idempotencyKey = "fx-probe:" + objectId, commentRequired = false, evidenceRequired = false
        });
        Check($"A (QA role) starts instance {objectId} → candidate {(candidate == aId ? "A" : "B")}", s is 200 or 201, $"{s} {Reason(b)}");
        return Find(b?["data"], "approvalTaskId") ?? throw new InvalidOperationException(b?.ToJsonString());
    }

    // 1) four eyes: A started, A is the assignee, A approves → 409 SOD_VIOLATION
    var t1 = await StartAs(aToken, "probe-1", aId);
    var (a1, a1b) = await Send(HttpMethod.Post, $"{PlatformUrl}/api/v1/workflow/tasks/{t1}/approve", aToken, new { reasonCode = "APPROVED", idempotencyKey = "fx-a-self" });
    Check("A approves own workflow", a1 == 409 && Reason(a1b) == "SOD_VIOLATION", $"{a1} {Reason(a1b)}");

    // 2) actor binding: A sends B's id in the body for B's task → 403 WORKFLOW_ACTOR_MISMATCH; B approves → 200
    var t2 = await StartAs(aToken, "probe-2", bId);
    var (a2, a2b) = await Send(HttpMethod.Post, $"{PlatformUrl}/api/v1/workflow/tasks/{t2}/approve", aToken, new { actorId = bId.ToString(), reasonCode = "APPROVED", idempotencyKey = "fx-a-as-b" });
    Check("A approves B's task naming B in the body", a2 == 403 && Reason(a2b) == "WORKFLOW_ACTOR_MISMATCH", $"{a2} {Reason(a2b)}");
    var (b2, b2b) = await Send(HttpMethod.Post, $"{PlatformUrl}/api/v1/workflow/tasks/{t2}/approve", bToken, new { reasonCode = "APPROVED", idempotencyKey = "fx-b" });
    Check("B approves B's task", b2 == 200, $"{b2} {Find(b2b, "newInstanceStatus")}");

    // the instance stores the starter's id; the approval log names the real person
    var pdb = mongo.GetDatabase("fx_probe_platform");
    var inst = await pdb.GetCollection<BsonDocument>("workflow_instances").Find(Builders<BsonDocument>.Filter.Eq("ObjectId", "probe-2")).FirstOrDefaultAsync();
    Check("instance StartedByUserId = A", inst is not null && string.Equals(inst.GetValue("StartedByUserId", BsonNull.Value).ToString(), aId.ToString(), StringComparison.OrdinalIgnoreCase),
        $"StartedBy={inst?.GetValue("StartedBy", BsonNull.Value)} StartedByUserId={inst?.GetValue("StartedByUserId", BsonNull.Value)}");

    // nothing was written to the shared server
    Check("probe never used port 27017", !runner.ConnectionString.Contains(":27017"), runner.ConnectionString.Split('?')[0]);
}
catch (Exception ex)
{
    Log("PROBE ERROR: " + ex);
    results.Add(("probe completed", false, ex.Message));
}
finally
{
    foreach (var p in processes) { try { if (!p.HasExited) p.Kill(entireProcessTree: true); p.WaitForExit(10000); } catch { } }
    runner.Dispose();
    Log($"teardown: {processes.Count} service processes stopped, probe mongod disposed");
    Log($"SUMMARY: {results.Count(r => r.Ok)}/{results.Count} checks passed");
    File.WriteAllText(Path.Combine(outDir, "probe-transcript.txt"), transcript.ToString());
}
return results.All(r => r.Ok) ? 0 : 1;
