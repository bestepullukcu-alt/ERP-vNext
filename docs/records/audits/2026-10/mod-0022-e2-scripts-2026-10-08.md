> Kanıt: `mod-0022-e2-evidence-2026-10-08.md` §5–§6. Depoya kod olarak eklenmedi (tek seferlik kanıt).
> Probe: iki dosyayı boş bir klasöre koyun, AuthService ve Platform API projelerini derleyin,
> `DITEN_ITEST_MONGOD_BIN_DIR` ayarlıyken `dotnet run -- <worktree> <çıktı klasörü>`. Kendi mongod'unu (replica set,

> Şifre ve biletler her koşuda rastgeledir ve hiçbir yere yazdırılmaz.

static byte[] Png(int width, int height)
{
    using var output = new MemoryStream();
    output.Write([0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A]);
    var header = new byte[13];
    System.Buffers.Binary.BinaryPrimitives.WriteInt32BigEndian(header.AsSpan(0), width);
    System.Buffers.Binary.BinaryPrimitives.WriteInt32BigEndian(header.AsSpan(4), height);
    header[8] = 8; header[9] = 2;
    Chunk(output, "IHDR", header);
    using var raw = new MemoryStream();
    var row = new byte[1 + width * 3];
    Array.Fill(row, (byte)0xFF);
    row[0] = 0;
    for (var y = 0; y < height; y++) { row[1 + (y % width) * 3] = 0x10; raw.Write(row); }
    using var compressed = new MemoryStream();
    using (var z = new System.IO.Compression.ZLibStream(compressed, System.IO.Compression.CompressionLevel.Optimal, true)) { raw.Position = 0; raw.CopyTo(z); }
    Chunk(output, "IDAT", compressed.ToArray());
    Chunk(output, "IEND", []);
    return output.ToArray();

    static void Chunk(Stream o, string type, byte[] data)
    {
        var len = new byte[4]; System.Buffers.Binary.BinaryPrimitives.WriteInt32BigEndian(len, data.Length); o.Write(len);
        var t = Encoding.ASCII.GetBytes(type); o.Write(t); o.Write(data);
        var crc = 0xFFFFFFFFu;
        foreach (var b in t.Concat(data)) { crc ^= b; for (var k = 0; k < 8; k++) crc = (crc & 1) != 0 ? (crc >> 1) ^ 0xEDB88320u : crc >> 1; }
        var c = new byte[4]; System.Buffers.Binary.BinaryPrimitives.WriteUInt32BigEndian(c, ~crc); o.Write(c);
    }
