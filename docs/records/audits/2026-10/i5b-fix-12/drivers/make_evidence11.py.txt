"""İ5B-FIX-12: the generated evidence texts from the raw outputs (no local paths).
Usage: make_evidence11.py <scratch ctfix09> <İ5B-FIX-12 evidence dir>"""
import hashlib, json, os, re, shutil, sys
S, I = sys.argv[1:3]
os.makedirs(os.path.join(I, "drivers"), exist_ok=True)


def sha(p):
    return hashlib.sha256(open(os.path.join(S, p), "rb").read()).hexdigest()


def write(name, lines):
    open(os.path.join(I, name), "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def rows_of(text):
    started = re.findall(r"(?:^|(?<=\.\.\. ))(test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\) \.\.\.", text, re.M)   # subtest lines run together
    fails = re.findall(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\)(?: (\(.*\)))?$", text, re.M)
    bad = {}
    for res, name, mod, cls, _ in fails:
        bad.setdefault(f"{mod}.{cls}.{name}", res)
    return ([(bad.get(f"{mod}.{cls}.{name}", "ok"), f"{mod}.{cls}.{name}") for name, mod, cls in started],
            [(res, name, sub) for res, name, mod, cls, sub in fails if sub])


freeze = dict(reversed(l.split()) for l in open(os.path.join(S, "i5b12-freeze-01.txt"), encoding="utf-8") if l.strip())

rows, subs = rows_of(text)

       "(check_instruction_budget), not the write / check CLI.", "",

out += [f"  {res:5s} {tid}" for res, tid in rows]
out += ["", "failed subtests:"] + [f"  {res:5s} {name} {sub}" for res, name, sub in subs]
out += ["", "The two new tests fail on the base source: the four marker-boundary cases pass there (core:start inside a",
        "reference or an inline title, a standalone core:start after a title opener, core:start and rule 1 on one line), and the",
        "escaped bang before a visible link is refused there (one and three backslashes). Their controls pass there too (core:start",
        "then a real blank line; a real image after zero or two backslashes; the rule only in a title). The 40 other tests pass in",
        "the same run.", "",
        "Earlier, not replaced: the two tests first ran red against the unchanged İ5B-FIX-11 source before the code was written",
        "(6 failures, the same cases)."]
write("red-first.txt", out)

       "İ5B-FIX-11's G-S3 is now G-P2); G-P1, G-P2, G-B1 are this package's.", ""]
for r in res:
    out += [r["mutation"] + f"  ({r['file']})", f"    tests: {', '.join(t.split('.')[-1] for t in r['tests'])}",
            f"    applied: {r['applied']} · parses: {r['parsed']} · named: {'RED' if r['red'] else 'green'} — {r['summary']}"
            f" · control: {r['control_summary']} · proven: {r['proven']}"]
