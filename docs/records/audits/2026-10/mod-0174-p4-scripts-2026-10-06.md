const run = crypto.randomBytes(2).toString('hex').toUpperCase();
const state = { run, tenant: crypto.randomUUID(), le: crypto.randomUUID(), user: crypto.randomUUID(), seeded: [] };
const PERMS = ['inventory.stock.read', 'inventory.movements.create', 'inventory.locations.read', 'inventory.locations.manage',

  'trace.lots.migrate', 'trace.serials.read'];

const b64 = (o) => Buffer.from(JSON.stringify(o)).toString('base64url');
function token(ttl = 7200) {
  const now = Math.floor(Date.now() / 1000);
  const body = b64({ alg: 'HS256', typ: 'JWT' }) + '.' + b64({
    sub: state.user, tenant_id: state.tenant, tenantId: state.tenant, legal_entity_id: state.le, actor_type: 'tenant_user',
    email: `e4-${run.toLowerCase()}@p4.test`, name: `E4 P4 ${run}`, permission: PERMS,
    iss: 'diten-auth-service', aud: 'diten-erp', nbf: now - 60, iat: now, exp: now + ttl
  });
  return body + '.' + crypto.createHmac('sha256', secret).update(body).digest('base64url');
}

async function call(method, path, body, headers = {}) {
  const res = await fetch(SERVICE + path, {
    method, headers: { 'content-type': 'application/json', authorization: 'Bearer ' + token(), ...headers },
    body: body === undefined ? undefined : JSON.stringify(body)
  });
  const text = await res.text();
  let json; try { json = text ? JSON.parse(text) : null; } catch { json = text; }
  return { status: res.status, etag: res.headers.get('etag') ?? undefined, body: json };
}
const key = (p) => `p4-${run}-${p}-${crypto.randomUUID().slice(0, 8)}`;
const post = (path, body) => call('POST', path, body, body.idempotencyKey ? { 'Idempotency-Key': body.idempotencyKey } : {});
const must = (r, what, status = 201) => { if (r.status !== status) throw new Error(`${what}: ${r.status} ${JSON.stringify(r.body)}`); return r.body; };

async function location(type, parent, code) {
  const body = { idempotencyKey: key('loc'), code, name: code, type };
  if (parent) body.parentId = parent;
  return must(await post('/api/locations', body), `location ${code}`).locationId;
}
function movement(quantity, lotId, to, status) {
  const b = { idempotencyKey: key('mv'), movementType: 'OPENING_BALANCE', itemId: ITEM, skuId: SKU, skuLevel: 'Lsku', quantity, uomId: 'EA',
    sourceModule: 'MOD-PROBE', sourceType: 'RUNTIME_PROBE', sourceDocumentId: `P4-${run}-DOC`, transactionDate: '2026-10-01T09:00:00Z',
    postingDate: '2026-10-01T09:00:00Z', toWarehouseId: state.wh, toLocationId: to, lotId };
  if (status) b.toStockStatus = status;
  return b;
}

async function seed() {
  const site = await location('Site', null, `SITE-${run}`);
  state.wh = await location('Warehouse', site, `WH-${run}`);
  const sl = await location('StorageLocation', state.wh, `SL-${run}`);
  state.binA = await location('Bin', sl, `A-${run}`);
  state.binB = await location('Bin', sl, `B-${run}`);

  const lot = must(await post('/api/trace/lots', { idempotencyKey: key('lot'), itemId: ITEM, lotNumber: `P4-${run}-EX`, skuId: SKU, skuLevel: 'Lsku' }), 'registered lot');
  state.registeredLot = lot.lotId;
  const typed = [
    ['exact (same number as the registered lot)', `P4-${run}-EX`, '10', state.binA],
    ['typo pair 1', `P4-${run}-L-001`, '6', state.binA],
    ['typo pair 2', `P4-${run}-L001`, '4', state.binB],
    ['typo pair 1, quality stock', `P4-${run}-L-001`, '2', state.binB, 'QUALITY_INSPECTION'],
    ['new lot', `P4-${run}-NEW`, '3', state.binA]
  ];
  for (const [label, text, qty, bin, status] of typed) {
    const r = must(await post('/api/inventory/movements', movement(qty, text, bin, status)), `typed ${text}`);
    state.seeded.push({ label, typedText: text, quantity: qty, transactionId: r.transactionId, transactionNumber: r.transactionNumber });
  }
  fs.writeFileSync(stateFile, JSON.stringify(state, null, 2));
}

const evidence = {
  groups: () => call('GET', '/api/trace/migration/manual-lots?pageSize=200'),
  status: () => call('GET', '/api/trace/migration/status'),
  links: () => call('GET', '/api/trace/migration/links'),
  lots: () => call('GET', `/api/trace/lots?skuId=${SKU}&skuLevel=Lsku&page=1&pageSize=50`),
  balances: () => call('GET', `/api/inventory/balance?skuId=${SKU}&page=1&pageSize=200`),
  transactions: () => call('GET', '/api/inventory/transactions?page=1&pageSize=200'),
  availability: () => call('GET', `/api/inventory/availability?skuId=${SKU}`),
  'typed-inbound-after': () => post('/api/inventory/movements', movement('1', `P4-${run}-LATE`, state.binA)),
  'lot-transactions': async (q) => call('GET', `/api/inventory/transactions?lotId=${encodeURIComponent(q.get('lotId') || state.registeredLot)}&page=1&pageSize=50`)
};

await seed();
http.createServer(async (req, res) => {
  const url = new URL(req.url, 'http://127.0.0.1:5197');
  try {
    if (url.pathname === '/login') {
      res.writeHead(302, {
        'Set-Cookie': `access_token=${token()}; Path=/; HttpOnly; SameSite=Lax`,
        Location: WEB + (url.searchParams.get('to') || '/LotMigration')
      });
      return res.end();
    }
    const name = url.pathname.replace(/^\/evidence\//, '');
    if (evidence[name]) {
      const r = await evidence[name](url.searchParams);
      res.writeHead(200, { 'content-type': 'application/json; charset=utf-8' });
      return res.end(JSON.stringify(r, null, 2));
    }
    res.writeHead(404); res.end();
  } catch (e) {
    res.writeHead(500, { 'content-type': 'text/plain' }); res.end(String(e.message || e));
  }
}).listen(5197, '127.0.0.1');
