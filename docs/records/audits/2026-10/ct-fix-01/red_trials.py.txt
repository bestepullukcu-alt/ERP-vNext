"""Red trials for CT-FIX-01 (package A, f5e656201) and I5B-FIX-01 (package B, c02b28c16), run BEFORE any source edit.
Production modules are loaded unmodified from the two fix worktrees; every broken input lives in a temp directory or a
temp git repository. A case is RED when today's code does the unsafe thing the audits reported.
Usage: python3 -I -B red_trials.py <worktree A> <worktree B> <temp dir>"""
import importlib.util
import json
import os
import subprocess
import sys
from pathlib import Path

sys.dont_write_bytecode = True
A, B, TMP = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3])
TMP.mkdir(parents=True, exist_ok=True)
results = []


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def case(cid, expected_safe, observed, red):
    results.append((cid, red))
    print(f"{'RED ' if red else 'ok  '} {cid}\n      expected (safe): {expected_safe}\n      observed       : {observed}")


ct = load("ct_queue", A / "scripts/ct_queue.py")
REG_HEADER, *REG_ROWS = (A / "execution/delivery/ct-queue-registry.tsv").read_text(encoding="utf-8").rstrip("\n").split("\n")
states_real = ct.read_table(ct.STATES, ct.STATE_COLUMNS)

# ---- 05a: the same queue registered twice: validate catches it, the dispatch path's loader does not
dup = TMP / "registry-dup.tsv"
dup.write_text("\n".join([REG_HEADER, REG_ROWS[0], REG_ROWS[0]]) + "\n", encoding="utf-8")
(TMP / "states.tsv").write_bytes(ct.STATES.read_bytes())
real_root, real_states = ct.ROOT, ct.STATES
ct.ROOT, ct.REGISTRY, ct.STATES = TMP, dup, TMP / "states.tsv"     # read_table names files relative to ROOT
reg_rows = ct.read_table(dup, ct.REGISTRY_COLUMNS)
validate_says = [p for p in ct.check_registry(reg_rows, states_real, commit_exists=lambda s: True, path_at=lambda s, p: True) if "twice" in p]
try:
    queues, _ = ct.load(reg_rows[0]["queue_id"])
    observed = f"load() returned {len(queues)} queue rows, no stop; require would use queues[0]"
    red = len(queues) > 1
except SystemExit as e:
    observed, red = f"load() stopped: {e}", False
case("05a duplicate registry row", "load() (used by require/guard/prompt gate) stops: DUR", f"{observed}; validate says: {validate_says}", red)
ct.ROOT, ct.REGISTRY, ct.STATES = real_root, A / "execution/delivery/ct-queue-registry.tsv", real_states

# ---- 05b: main registers the queue strict=yes, the branch strict=no: protection silently lowered
cols = REG_HEADER.split("\t")
row = dict(zip(cols, REG_ROWS[0].split("\t")))
branch_q = dict(row, strict="no", vocabulary="sop7")
main_text = "\n".join([REG_HEADER, "\t".join(dict(row, strict="yes", vocabulary="sop7")[c] for c in cols)]) + "\n"
status, detail = ct.in_force_status(branch_q, main_text, commit_exists=lambda s: True, path_at=lambda s, p: True)
case("05b main strict=yes, branch strict=no", "in_force_status: not 'kayıtlı' (profile mismatch)", f"'{status}' {detail}", status == "kayıtlı")

# ---- 06a: a state defined twice; the second row makes FAILED dispatchable
dup_states = states_real + [dict(s, dispatch="yes") for s in states_real if s["queue_id"] == "sop7" and s["state"] == "FAILED"]
vocab = ct.vocabulary(dup_states, "sop7")
validate_says = [p for p in ct.check_registry([], dup_states) if "twice" in p]
case("06a FAILED listed twice, second row dispatch=yes", "the vocabulary is refused (DUR) before any dispatch decision",
     f"vocabulary()['FAILED']['dispatch'] = '{vocab['FAILED']['dispatch']}'; validate says: {validate_says}", vocab["FAILED"]["dispatch"] == "yes")

# ---- 06b: one transition row CANCELLED -> READY re-opens a terminal state
vocab = ct.vocabulary(states_real, "sop7")
edges = ct.load_transitions("sop7")
term_before = ct.terminal_states(vocab, edges)
term_after = ct.terminal_states(vocab, edges | {("CANCELLED", "READY")})
hdr = "id\titem\tstate\towner\tdepends_on\tredispatch_trigger\tsince\trecord"
v1 = ct.Table(hdr + "\nW3\tx\tCANCELLED\tCT\t-\t-\t2026-10-01\t-\n")
v2 = ct.Table(hdr + "\nW3\tx\tREADY\tCT\t-\t-\t2026-10-01\t-\n")
exits_before = ct.terminal_exits([v1, v2], term_before)
exits_after = ct.terminal_exits([v1, v2], term_after)
case("06b transition table gains CANCELLED->READY", "CANCELLED stays terminal (card §0: ACCEPTED, REWORK, FAILED, SUPERSEDED, CANCELLED)",
     f"terminal before {sorted(term_before)}; after {sorted(term_after)}; exit errors before {len(exits_before)}, after {len(exits_after)}",
     "CANCELLED" not in term_after)

# ---- 09: one module registry row cut to 5 columns disappears; discovery says "no match"
vmi = load("verify_module_id", A / ".antigravity/scripts/verify_module_id.py")
reg_text = (A / vmi.REGISTRY).read_text(encoding="utf-8")
target = next(l for l in reg_text.split("\n") if l.startswith("| MOD-0009-FU02 "))
cut = "| " + " | ".join([c.strip() for c in target.strip().strip("|").split("|")][:5]) + " |"
bad = TMP / "module-id-registry-cut.md"
bad.write_text(reg_text.replace(target, cut), encoding="utf-8")
_, code_good = vmi.discover(str(A), "Tenant Manager Backend")
vmi.REGISTRY = str(bad)                      # os.path.join(root, absolute) -> the temp copy
lines_bad, code_bad = vmi.discover(str(A), "Tenant Manager Backend")
vmi.REGISTRY = "execution/registries/module-id-registry.md"
case("09a MOD-0009-FU02 row cut to 5 columns", "discovery: 2 (tamamlanamadı) naming the bad line",
     f"intact registry exit {code_good}; cut registry exit {code_bad}; last line: {lines_bad[-1][:90] if lines_bad else '-'}", code_bad == 0)
cells = [c.strip() for c in target.strip().strip("|").split("|")]
blank = "| " + " | ".join([""] + cells[1:]) + " |"
bad.write_text(reg_text.replace(target, blank), encoding="utf-8")
vmi.REGISTRY = str(bad)
lines_blank, code_blank = vmi.discover(str(A), "Tenant Manager Backend")
vmi.REGISTRY = "execution/registries/module-id-registry.md"
case("09b MOD-0009-FU02 row with an empty id cell (9 columns)", "discovery: 2 (tamamlanamadı) naming the bad line",
     f"exit {code_blank}; last line: {lines_blank[-1][:90] if lines_blank else '-'}", code_blank == 0)

# ---- 11: pack and governance status
pg = load("prompt_gate", A / "scripts/prompt_gate.py")


class FakeRepo:
    def __init__(self, files): self.f = files
    def file(self, commit, path): return self.f[path].encode() if path in self.f else None


pack = "execution/domains/x/module-packs/MOD-0001-x.md"
for cid, text in [("11a pack: status approved then draft", "---\nid: MOD-0001\nstatus: approved\nstatus: draft\n---\nbody\n"),
                  ("11b pack: status only outside a front matter", "# MOD-0001\nstatus: approved\nbody\n"),
                  ("11c pack: front matter never closed", "---\nid: MOD-0001\nstatus: approved\nbody\n")]:
    eng = pg.check_authority({"authority": pack, "task_class": "geliştirme", "wp": "W1"}, FakeRepo({pack: text}), "c" * 40)
    case(cid, "ENGEL (one closed front matter with exactly one status)", f"engels: {eng}", not eng)
rec = "docs/records/decisions/2026-10/x.md"
for cid, text in [("11d governance: approved then draft", "---\nstatus: approved\nstatus: draft\n---\nW1\n"),
                  ("11e governance: approval only in prose", "# Karar\nstatus: approved\nW1\n")]:
    eng = pg.check_authority({"authority": rec, "task_class": "yönetişim", "wp": "W1"}, FakeRepo({rec: text}), "c" * 40)
    case(cid, "ENGEL (one closed front matter with exactly one status)", f"engels: {eng}", not eng)

# ---- 13: a hex-looking BRANCH name resolves as if it were a commit
repo = TMP / "hexrepo"
if not repo.exists():
    env = dict(os.environ, GIT_AUTHOR_NAME="t", GIT_AUTHOR_EMAIL="t@invalid", GIT_COMMITTER_NAME="t",
               GIT_COMMITTER_EMAIL="t@invalid", GIT_AUTHOR_DATE="2026-10-17T00:00:00Z", GIT_COMMITTER_DATE="2026-10-17T00:00:00Z")
    for args in (["init", "-q", str(repo)], ["-C", str(repo), "commit", "-q", "--allow-empty", "-m", "x"]):
        subprocess.run(["git", *args], check=True, env=env)
sha = subprocess.run(["git", "-C", str(repo), "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()
name = next(n for n in ("deadbee", "deadbef", "cafe123") if not sha.startswith(n))
subprocess.run(["git", "-C", str(repo), "branch", "-f", name, sha], check=True)
ar = load("acceptance_record", A / "scripts/acceptance_record.py")
sr2 = load("status_report_v2", A / "scripts/status_report_v2.py")
got_ar = ar.Git(repo).resolve(name)
got_pg = pg.GitRepo(str(repo)).resolve(name)
sr2.ROOT = repo
got_sr2 = sr2.Source.resolve(object.__new__(sr2.Source), name)
for tool, got in (("acceptance_record", got_ar), ("prompt_gate", got_pg), ("status_report_v2", got_sr2)):
    case(f"13 {tool}: branch '{name}' (commit {sha[:9]})", "None (a hex id must be a commit with that prefix)", f"resolved to {got}", got is not None)

     f"problems {problems} (with the file present: {base_problems})", not any("AGENTS-REFERENCE" in p for p in problems))
