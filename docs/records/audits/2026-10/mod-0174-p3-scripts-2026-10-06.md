## run-web-p3.ps1 — own Diten.Web on 127.0.0.1:5195

```powershell
<#
  WP-0174-P3 E4 — OWN Diten.Web instance (worktree Release build) on 127.0.0.1:5195 (--urls; never the owner's 5001),
  environment Development, GatewayUrl = the owner's Gateway 5000 (appsettings.Development), PRODUCT-MASTER reads of the
  Web through the PM0 dev bridge to the owner's Prism 4010 (Inventory__ProductMasterBaseUrl, environment only).
  127.0.0.1 (not localhost): its cookies never mix with the owner's localhost:5001 session.
  -Action Start | Stop. Records the owner's fleet PIDs before / after. Nothing of the owner's is stopped or restarted.
#>
param([ValidateSet('Start', 'Stop')][string]$Action = 'Start', [string]$Root = $PSScriptRoot, [switch]$WithoutTenantLiveness)
$ErrorActionPreference = 'Stop'
$web = 'C:\Users\Ali\erp-vnext-wt-0174-p3\frontend\Diten.Web'
$state = Join-Path $Root 'web-state.json'
$fleetPorts = 4010, 5000, 5001, 5056, 5057, 5059, 5066, 27017
function Fleet { foreach ($p in $fleetPorts) { $c = Get-NetTCPConnection -LocalPort $p -State Listen -ErrorAction SilentlyContinue | Select-Object -First 1; "$p=$($c.OwningProcess)" } }

if ($Action -eq 'Stop') {
    $s = Get-Content $state -Raw | ConvertFrom-Json
    & taskkill /PID $s.pid /T /F 2>$null | Out-Null
    Start-Sleep -Seconds 2
    [pscustomobject]@{ stopped = $s.pid; listenerLeftOn5195 = [bool](Get-NetTCPConnection -LocalPort 5195 -State Listen -ErrorAction SilentlyContinue); fleetBefore = $s.fleetBefore; fleetAfter = @(Fleet) } | ConvertTo-Json
    return
}

if (Get-NetTCPConnection -LocalPort 5195 -State Listen -ErrorAction SilentlyContinue) { throw 'Port 5195 is in use.' }
$fleetBefore = @(Fleet)
$vars = @{ ASPNETCORE_ENVIRONMENT = 'Development'; ASPNETCORE_URLS = 'http://127.0.0.1:5195'; Kestrel__Endpoints__Http__Url = 'http://127.0.0.1:5195'; Inventory__ProductMasterBaseUrl = 'http://127.0.0.1:4010/' }
if ($WithoutTenantLiveness) { $vars['Platform__InternalApiKey'] = ' ' }
foreach ($k in $vars.Keys) { Set-Item "Env:$k" $vars[$k] }
try {
    $p = Start-Process dotnet -ArgumentList "`"$web\bin\Release\net8.0\Diten.Web.dll`"", '--urls', 'http://127.0.0.1:5195' -WorkingDirectory $web -PassThru -WindowStyle Hidden `
        -RedirectStandardOutput (Join-Path $Root 'web.log') -RedirectStandardError (Join-Path $Root 'web.err.log')
} finally { foreach ($k in $vars.Keys) { Remove-Item "Env:$k" -ErrorAction SilentlyContinue } }
$deadline = (Get-Date).AddSeconds(120)
while (-not (Get-NetTCPConnection -LocalPort 5195 -State Listen -ErrorAction SilentlyContinue)) { if ((Get-Date) -gt $deadline -or $p.HasExited) { throw 'Web did not open 5195.' }; Start-Sleep -Milliseconds 500 }
$bound = (Get-NetTCPConnection -State Listen | Where-Object OwningProcess -eq $p.Id | ForEach-Object { "$($_.LocalAddress):$($_.LocalPort)" }) -join ','
[pscustomobject]@{ pid = $p.Id; bound = $bound; withoutTenantLiveness = [bool]$WithoutTenantLiveness; fleetBefore = $fleetBefore } | ConvertTo-Json | Set-Content $state -Encoding utf8
Get-Content $state -Raw
```

Used as `-Action Start` (first login → `tenant-unavailable`), then `-Action Stop` + `-Action Start -WithoutTenantLiveness`
(own instance only), restarted after each Release rebuild, finally `-Action Stop` (output = fleet before / after).
