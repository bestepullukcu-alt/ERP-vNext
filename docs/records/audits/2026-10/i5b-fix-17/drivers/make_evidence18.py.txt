def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def write(name, lines):
    open(os.path.join(E, name), "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def pairs(path):
    return dict(reversed(l.split()) for l in open(path, encoding="utf-8") if l.strip())

out = ["İ5B-FIX-17 red first — this package's final test file in the İ5B worktree while sync_adapters.py was still C15's",
       "(eca57404d, İ5B-FIX-16), before the code was written; run by the AI session. The two files run (SHA-256):"]
out += [f"  {p:50s}{h}" for p, h in inputs.items()]
out += [f"  the test equals the frozen one: {inputs['scripts/tests/test_sync_adapters_budget.py'] == freeze['scripts/tests/test_sync_adapters_budget.py']}", "",
        re.search(r"^Ran .*$", text, re.M).group(0) + " · " + re.search(r"^(OK|FAILED).*$", text, re.M).group(0),

        "'not verified'; Codex F2 is refused as a script never closed; the type-7 line with a quoted '>' is a false refusal. The",
        "quoted '>'-only hidden case and the four harmless controls already pass on C15 and guard this package's own rules."]

runs = [("01", "harness17.py", "on the first frozen source 16a11a3a (with the indented-code check)"),
        ("02", "harness17b.py", "on the final source 0e456b0f (the indented-code check removed as redundant)")]

       "mutation (rsync from the İ5B-FIX-14 base, the two frozen files laid over it, checked by SHA-256); run by the AI session.",

       "anchors are all still once in the final source and their code is untouched, so they are not run again.", ""]
