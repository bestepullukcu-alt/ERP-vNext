# WP-0176-F3 runtime probe — the built InventoryService on ITS OWN port (5167) with ITS OWN mongod (27417) and an isolated database.

$ErrorActionPreference = 'Stop'
$scratch = Split-Path -Parent $MyInvocation.MyCommand.Path
$repo = 'C:\Users\Ali\erp-vnext-wt-0176'
$mongod = 'C:\Program Files\MongoDB\Server\9.0\bin\mongod.exe'

$fleetPorts = @(4010, 5000, 5001, 5056, 5057, 5058, 5059, 5060, 5061, 5066, 5067, 27017)

New-Item -ItemType Directory -Force $out | Out-Null
$dbPath = Join-Path $out 'mongo-data'
New-Item -ItemType Directory -Force $dbPath | Out-Null
$result = [ordered]@{}

function Fleet {
    $rows = [ordered]@{}
    foreach ($p in $fleetPorts) {
        $c = Get-NetTCPConnection -State Listen -LocalPort $p -ErrorAction SilentlyContinue | Select-Object -First 1

    return $rows

    -RedirectStandardOutput $svcLog -RedirectStandardError (Join-Path $out 'service.err.log') -PassThru -WindowStyle Hidden

try {
    $deadline = (Get-Date).AddSeconds(60)

        Start-Sleep -Seconds 1
        try { $health = Invoke-WebRequest -Uri "http://127.0.0.1:$svcPort/health" -UseBasicParsing -TimeoutSec 3 } catch { $health = $null }
    } until ($health -or (Get-Date) -gt $deadline)
    $result.health = if ($health) { "$($health.StatusCode) $($health.Content)" } else { 'no answer' }

    function Probe($method, $path) {
