def git(*a) -> str:
    return subprocess.run(["git", "-C", str(C), *a], capture_output=True, text=True, env=ENV, check=True).stdout.strip()


def write(rel: str, text: str):
    p = C / rel
    p.parent.mkdir(parents=True, exist_ok=True)  # inside the scratch clone only
    p.write_text(text, encoding="utf-8")


def commit(msg: str, *paths: str) -> str:
    git("add", "--", *paths)
    git("commit", "-q", "-m", msg)
    return git("rev-parse", "HEAD")


def expect(cond: bool, what: str):
    log.append(f"   {'PASS' if cond else 'FAIL'} expected: {what}")
    if not cond:
        failures.append(what)


def run(*args: str, script: str = "scripts/acceptance_record.py") -> tuple[int, str]:
    done = subprocess.run([sys.executable, script, *args], cwd=C, capture_output=True, text=True, env=ENV)
    return done.returncode, done.stdout + (("\n[stderr]\n" + done.stderr) if done.stderr.strip() else "")


def mc(label: str, pack: str, *extra: str, module: str = M, queue: str = Q) -> tuple[int, str]:
    code, out = run("module-check", "--queue", queue, "--module", module, "--pack", pack, *extra)
    log.append(f"## {label} — exit {code}")
    log.extend("   " + l[:200] for l in out.splitlines() if l.startswith(("İŞ", "AC-", "sonuç", "paket:")))
    return code, out

def scenario(name: str, records: dict[str, str], rows: list[tuple]) -> str:
    """A branch from BASE: commit the records, then the queue rows that point at them."""
    git("checkout", "-q", "-B", f"mc-{name}", BASE)
    for path, text in records.items():
        write(path, text)
    cr = commit(f"e2e {name}: records", *records) if records else BASE
    body = "".join("\t".join([wid, "AC işi", mod, state, "CT-X", dep, "-", "2026-10-16",
                              (f"{rec}@{cr}" if rec != "-" else "-"), "-"]) + "\n" for wid, mod, state, dep, rec in rows)
    write(QPATH, tpl + body)
    return commit(f"e2e {name}: queue", QPATH)
