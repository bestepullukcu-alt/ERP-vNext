"""CT-FIX-06 / İ5B-FIX-07: the generated evidence texts from the raw outputs (no local paths in the output).
Usage: make_evidence6.py <scratch ctfix06> <CT evidence dir> <İ5B evidence dir>"""
import json, re, sys, os, shutil, collections
S, ECT, EI = sys.argv[1:4]
for d in (ECT, EI, os.path.join(ECT, "drivers"), os.path.join(EI, "drivers")):
    os.makedirs(d, exist_ok=True)

def red(raw, prefix):
    text = open(raw, encoding="utf-8").read()
    ran = re.search(r"^Ran (\d+) tests", text, re.M).group(1)
    per = collections.Counter()
    for m in re.finditer(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\)", text, re.M):
        per[f"{m.group(3)}.{m.group(4)}.{m.group(2)}" + (" (error)" if m.group(1) == "ERROR" else "")] += 1
    return ran, per

def block(title, parts):
    out = [title, ""]
    for label, (ran, per) in parts:
        out += [f"{label}: ran {ran} tests · failed subtests {sum(per.values())} in {len(per)} test methods"]
        out += [f"  {n:4d}  {k}" for k, n in sorted(per.items())] + [""]
    return "\n".join(out)

ct_note = ("Base: the committed sources of c203303ce (CT-FIX-05) in a copy outside the repository, this package's test files\n"
           "copied in; run by the AI session. Every method not listed passed. The two errors of the real-registry tests in the\n"
           "first run are the copy's environment (it holds no execution/registries), not a red of this package.")
open(os.path.join(ECT, "red-first.txt"), "w", encoding="utf-8").write(block(
    "CT-FIX-06 red first — the new tests against the unchanged base code.\n" + ct_note,

     ("evidence lines and required registry columns (R0 CTFIX06_EvidenceLineAndRegistry)", red(os.path.join(S, "red-first/ct-red-2.txt"), ""))]))
open(os.path.join(EI, "red-first.txt"), "w", encoding="utf-8").write(block(
    "İ5B-FIX-07 red first — this package's test_sync_adapters_budget.py against the unchanged base code 5cd4cc527\n"
    "(İ5B-FIX-06) in a copy outside the repository; run by the AI session. The 25 errors are the real-file measurement,\n"
    "which now calls sync_adapters.on_rule_line (absent in the base). Every method not listed passed.",
    [("test_sync_adapters_budget", red(os.path.join(S, "red-first/i5b-red.txt"), ""))]))

def guards(files, title, note):
    out = [title, note, "Each mutation: the file still parses; the named tests fail; the control test passes in the same copy.", ""]
    total = proven = 0
    for js in files:
        for r in json.load(open(js, encoding="utf-8")):
            total += 1; proven += r["proven"]
            out += [r["mutation"], f"    file: {r['file']} · tests: {', '.join(t.split('.')[-1] for t in r['tests'])}",
                    f"    named: {'RED' if r['red'] else 'green'} — {r['summary']} · control: {r['control_summary']} · proven: {r['proven']}"]
    return "\n".join(out + ["", f"proven: {proven} of {total}"]) + "\n"
