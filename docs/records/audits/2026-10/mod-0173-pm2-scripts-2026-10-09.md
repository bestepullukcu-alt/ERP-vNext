> Kanıt: `mod-0173-pm2-evidence-2026-10-09.md` §4–§6. Tek seferlik çalışma zamanı kanıtı (WP-0023-FX yöntemi), repo kodu değil.

    await Move(le1, wh1, bin1, skuB, "GOODS_RECEIPT_PO", "3", to: "BLOCKED");
    await Move(le2, wh2, bin2, skuT, "GOODS_RECEIPT_PO", "2", to: "IN_TRANSIT");          // only in the OTHER company, only in transit
    await Move(le1, wh1, bin1, skuI, "GOODS_RECEIPT_PO", "6", to: "QUALITY_INSPECTION");
    await Move(le1, wh1, bin1, skuZ, "OPENING_BALANCE", "5", to: "AVAILABLE");
    await Move(le1, wh1, bin1, skuZ, "GOODS_ISSUE_SALES", "5", from: "AVAILABLE");           // back to zero
    await Move(le2, wh2, bin2, skuR, "OPENING_BALANCE", "8", to: "AVAILABLE");
    results["movements"] = movements;

    // DB read: the non-zero balance rows per SKU, company and status, straight from inventory_balances.
    var balances = mongo.GetDatabase("itest_pm2_inventory").GetCollection<BsonDocument>("inventory_balances");
    var dbRows = new JsonArray();
    foreach (var (name, sku) in new[] { ("Q", skuQ), ("B", skuB), ("T", skuT), ("I", skuI), ("Z", skuZ), ("N", skuN), ("R", skuR) })
    {
        var rows = await balances.Find(new BsonDocument { { "TenantId", tenantId.ToString() }, { "SkuId", sku.ToString() } }).ToListAsync();
        dbRows.Add($"sku {name}: " + (rows.Count == 0 ? "no row" : string.Join("; ", rows.Select(r =>
            $"{(r["LegalEntityId"].AsString == le1.ToString() ? "LE1" : "LE2")} {r["StockStatus"].AsString} onHand={r["OnHand"].ToDecimal()} deleted={r["IsDeleted"].AsBoolean}"))));
    }
    results["dbRows"] = dbRows;
    foreach (var line in dbRows) Log("db: " + line);

    // ── 7. the operation ─────────────────────────────────────────────────────────────────────────────────
    string Key(string id, string secret) => $"{id}:{secret}";
    async Task<(int Status, JsonNode? Body, string Raw, long Ms)> Presence(Guid sku, string level = "Lsku", string? key = null, Guid? tenant = null, string? token = null, string url = InventoryUrl)
        => await Send(HttpMethod.Get, $"{url}/api/internal/inventory/stock-presence?skuId={sku}&skuLevel={level}", token, null,
            internalApiKey: key ?? Key("ditenmdmservice", mdm1), tenantHeader: (tenant ?? tenantId).ToString());
    async Task Expect(string name, Guid sku, bool expected, string level = "Lsku", Guid? tenant = null, string? secret = null)
    {
        var (s, b, raw, ms) = await Presence(sku, level, secret is null ? null : Key("ditenmdmservice", secret), tenant);
        Check(name, s == 200 && b?["hasNonZeroStock"]?.GetValue<bool>() == expected && b?["coverage"]?.ToString() == "COMPLETE"
            && b?.AsObject().Count == 6, $"{s} {raw} ({ms} ms)");
    }
    await Expect("never-stocked SKU → 200 false", skuN, false);
    await Expect("stock only in QUARANTINE → true", skuQ, true);
    await Expect("stock only in BLOCKED → true", skuB, true);
    await Expect("stock only IN_TRANSIT, only in the OTHER company of the tenant → true", skuT, true);
    await Expect("stock only in QUALITY_INSPECTION → true", skuI, true);
    await Expect("balance back to zero → false", skuZ, false);
    await Expect("AVAILABLE stock in company 2 → true", skuR, true);
    var otherTenant = Guid.NewGuid();
    await Expect("ANOTHER tenant asking for the same SKUs → false (Q)", skuQ, false, tenant: otherTenant);
    await Expect("ANOTHER tenant asking for the same SKUs → false (R)", skuR, false, tenant: otherTenant);
    await Expect("Item level accepted (nothing stocked at Item) → false", skuQ, false, level: "Item");
    await Expect("second secret of the rotation overlap → 200", skuQ, true, secret: mdm2);

    var refusals = new List<(string Name, (int Status, JsonNode? Body, string Raw, long Ms) R)>
    {
        ("no key", await Send(HttpMethod.Get, $"{InventoryUrl}/api/internal/inventory/stock-presence?skuId={skuQ}&skuLevel=Lsku", null, null)),
        ("wrong secret", await Presence(skuQ, key: Key("ditenmdmservice", Convert.ToBase64String(RandomNumberGenerator.GetBytes(32))))),
        ("unknown identifier", await Presence(skuQ, key: Key("ditenunknown", mdm1))),
        ("valid key of another operation", await Presence(skuQ, key: Key("otherservice", other))),
        ("user token only", await Send(HttpMethod.Get, $"{InventoryUrl}/api/internal/inventory/stock-presence?skuId={skuQ}&skuLevel=Lsku", aToken, null)),
        ("user token + valid key", await Presence(skuQ, token: aToken))
    };
    var firstBody = refusals[0].R.Raw;
    foreach (var (name, r) in refusals)
        Check($"refusal: {name} → 401, body identical", r.Status == 401 && r.Raw == firstBody, $"{r.Status} {r.Raw}");
    var bad = await Presence(skuQ, level: "lsku");
    Check("unknown skuLevel → 400 INVALID_REQUEST", bad.Status == 400 && bad.Body?["error"]?["code"]?.ToString() == "INVALID_REQUEST", $"{bad.Status} {bad.Raw}");
    var noTenant = await Send(HttpMethod.Get, $"{InventoryUrl}/api/internal/inventory/stock-presence?skuId={skuQ}&skuLevel=Lsku", null, null,
        internalApiKey: Key("ditenmdmservice", mdm1), tenantHeader: "not-a-uuid");
    Check("X-Tenant-Id not a UUID → 400", noTenant.Status == 400, $"{noTenant.Status} {noTenant.Raw}");

    // https rule: the Staging instance refuses plain http even with the right key (same 401) and logs a warning.
    var staging = await Presence(skuQ, url: StagingUrl);
    await Task.Delay(1000);
    string stagingLog;
    using (var fs = new FileStream(Path.Combine(outDir, "stack-inventory-staging.log"), FileMode.Open, FileAccess.Read, FileShare.ReadWrite))
        stagingLog = new StreamReader(fs).ReadToEnd();
    Check("Staging (not Development) over plain http, right key → the same 401 + warning log", staging.Status == 401 && staging.Raw == firstBody
        && stagingLog.Contains("refused: the call is not https"), $"{staging.Status} {staging.Raw}");

    // p95: 300 calls on the Development instance (mixed present / absent / other tenant).
    var times = new List<long>();
    var pool = new[] { skuQ, skuB, skuT, skuI, skuZ, skuN, skuR };
    for (var i = 0; i < 300; i++)
    {
        var r = await Presence(pool[i % pool.Length], tenant: i % 5 == 0 ? otherTenant : tenantId);
        if (r.Status != 200) throw new InvalidOperationException("p95 run: " + r.Status);
        times.Add(r.Ms);
    }
    times.Sort();
    long P(double q) => times[(int)Math.Ceiling(q * times.Count) - 1];
    results["latencyMs"] = new JsonObject { ["calls"] = times.Count, ["p50"] = P(0.50), ["p95"] = P(0.95), ["p99"] = P(0.99), ["max"] = times[^1] };
    Check("p95 < 500 ms (300 calls, own data, Development host)", P(0.95) < 500, $"p50 {P(0.50)} ms, p95 {P(0.95)} ms, p99 {P(0.99)} ms, max {times[^1]} ms");

    // Rotation live: the script retires the oldest secret (mdm1) — no restart; the running service reads its configuration per call.
    await Script("rotate again with -RetireOldest (drops the first secret)", "-RetireOldest");
    var afterRotation = ReadSecrets("ditenmdmservice");
    secrets.AddRange(afterRotation.Where(s => !secrets.Contains(s)));
    var oldRefused = false;
    for (var i = 0; i < 30 && !oldRefused; i++) { oldRefused = (await Presence(skuQ)).Status == 401; if (!oldRefused) await Task.Delay(1000); }
    var keptOk = (await Presence(skuQ, key: Key("ditenmdmservice", mdm2))).Status == 200;
    var newOk = (await Presence(skuQ, key: Key("ditenmdmservice", afterRotation.Single(s => s != mdm2)))).Status == 200;
    Check("rotation without restart: the retired secret → 401; the kept and the new one → 200", oldRefused && keptOk && newOk,
        $"retired 401={oldRefused}, kept 200={keptOk}, new 200={newOk}; secrets now {afterRotation.Count}");

    // ── 8. audit + logs ──────────────────────────────────────────────────────────────────────────────────
    var audit = mongo.GetDatabase("itest_pm2_inventory").GetCollection<BsonDocument>("inventory_audit");
    var rowsA = await audit.Find(new BsonDocument("Action", "INTERNAL_STOCK_PRESENCE_READ")).ToListAsync();
    results["audit"] = new JsonObject
    {
        ["rows"] = rowsA.Count,
        ["byOutcome"] = new JsonObject(rowsA.GroupBy(r => r["Outcome"].AsString).Select(g => KeyValuePair.Create(g.Key, (JsonNode?)g.Count()))),
        ["byKey"] = new JsonObject(rowsA.GroupBy(r => r["KeyIdentifier"].IsBsonNull ? "(none)" : r["KeyIdentifier"].AsString).Select(g => KeyValuePair.Create(g.Key, (JsonNode?)g.Count()))),
        ["sample200"] = rowsA.First(r => r["StatusCode"].AsInt32 == 200).ToJson(new MongoDB.Bson.IO.JsonWriterSettings { OutputMode = MongoDB.Bson.IO.JsonOutputMode.RelaxedExtendedJson }),
        ["sample401"] = rowsA.First(r => r["StatusCode"].AsInt32 == 401).ToJson(new MongoDB.Bson.IO.JsonWriterSettings { OutputMode = MongoDB.Bson.IO.JsonOutputMode.RelaxedExtendedJson })
    };
    var auditText = string.Join("\n", rowsA.Select(r => r.ToJson()));
    Check("audit: every call has a row; the 200s carry ditenmdmservice; tenant, sku, level, result, correlation present",
        rowsA.Count >= 300 && rowsA.Where(r => r["StatusCode"].AsInt32 == 200).All(r => r["KeyIdentifier"].AsString == "ditenmdmservice"
            && !r["TenantId"].IsBsonNull && !r["SkuId"].IsBsonNull && !r["SkuLevel"].IsBsonNull && !r["CorrelationId"].IsBsonNull),
        $"{rowsA.Count} rows");
    await Task.Delay(1500);
    var logText = string.Join("\n", Directory.GetFiles(outDir, "stack-*.log").Select(f =>
    {
        using var fs = new FileStream(f, FileMode.Open, FileAccess.Read, FileShare.ReadWrite);
        return new StreamReader(fs).ReadToEnd();
    }));
    var transcript = File.ReadAllText(Path.Combine(outDir, "transcript.txt"));
    var leaks = secrets.Sum(s => (logText.Contains(s) ? 1 : 0) + (auditText.Contains(s) ? 1 : 0) + (transcript.Contains(s) ? 1 : 0));
    Check("secret scan: logs of every probe process, the audit rows and this transcript", leaks == 0, $"{secrets.Count} secrets × 3 sources: {leaks} matches");
    Check("key identifier is in the audit", auditText.Contains("ditenmdmservice"), "ditenmdmservice present");

    results["userSecretsId"] = userSecretsId;
    results["pids"] = new JsonArray(processes.Select(p => (JsonNode)JsonValue.Create(p.Id)!).ToArray());
    results["mongodPort"] = mongoUrl.Servers.Single().Port;
    File.WriteAllText(Path.Combine(outDir, "results.json"), results.ToJsonString(new JsonSerializerOptions { WriteIndented = true, Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping, TypeInfoResolver = new System.Text.Json.Serialization.Metadata.DefaultJsonTypeInfoResolver() }));
    Log($"DONE: {checks.Count(c => c!["ok"]!.GetValue<bool>())}/{checks.Count} checks OK");
}
catch (Exception ex)
{

    File.WriteAllText(Path.Combine(outDir, "results.json"), results.ToJsonString(new JsonSerializerOptions { WriteIndented = true, TypeInfoResolver = new System.Text.Json.Serialization.Metadata.DefaultJsonTypeInfoResolver() }));

{
    var pids = processes.Select(p => p.Id).ToList();
    foreach (var p in processes) { try { if (!p.HasExited) p.Kill(entireProcessTree: true); p.WaitForExit(10000); } catch { } }
    runner.Dispose();
    try { Directory.Delete(appData, recursive: true); } catch { }

}
