"""İ5B-FIX-18: the generated evidence texts from the raw outputs (no local paths).
Usage: make_evidence19.py <scratch ctfix10> <İ5B-FIX-18 evidence dir>"""
import hashlib, json, os, re, shutil, sys
S, E = sys.argv[1:3]
T = os.path.join(S, "i5b18")
os.makedirs(os.path.join(E, "drivers"), exist_ok=True)


def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def write(name, lines):
    open(os.path.join(E, name), "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def pairs(path):
    return dict(reversed(l.split()) for l in open(path, encoding="utf-8") if l.strip())


freeze = pairs(os.path.join(T, "i5b18-freeze-01.txt"))
NEW = "test_a_closing_tag_has_the_commonmark_form"
out = ["İ5B-FIX-18 red first — this package's test file in the İ5B worktree while sync_adapters.py was still C16's (2de454f49,",
       "İ5B-FIX-17), before the code was written; run by the AI session.", ""]
for run, label in (("red-i5b18-pre.txt", "run 01 (kept: one positive was wrong — see below)"), ("red-i5b18-pre-02.txt", "run 02 (the final test file)")):
    text = open(os.path.join(T, run), encoding="utf-8").read()
    inputs = pairs(os.path.join(T, "red-inputs-01.txt" if run.endswith("pre.txt") else "red-inputs-02.txt"))
    fails = re.findall(r"^FAIL: (test_\w+) \(scripts\.tests\.\w+\.\w+\)(?: (\(.*\)))?$", text, re.M)
    out += [f"{label}:"] + [f"  {p:50s}{h}" for p, h in inputs.items()]
    out += ["  " + re.search(r"^Ran .*$", text, re.M).group(0) + " · " + re.search(r"^(OK|FAILED).*$", text, re.M).group(0),
            f"  every failure is in the new test: {all(n == NEW for n, _ in fails)}; the 53 earlier tests pass"]
    out += [f"    FAIL  {sub}" for n, sub in fails]
    out.append("")
out += [f"the final test equals the frozen one: {pairs(os.path.join(T, 'red-inputs-02.txt'))['scripts/tests/test_sync_adapters_budget.py'] == freeze['scripts/tests/test_sync_adapters_budget.py']}", "",
        "Run 01 held a mistake in the test, not in the code: its positive '</div\\n>' put '>' at the start of the next line,",
        "and a line that starts with '>' is a block quote in CommonMark, so it ends the paragraph — not a valid closing tag.",
        "The positive was changed to '</div\\n    >' (four spaces: a paragraph continuation) before the code was written."]
write("red-first.txt", out)

res = json.load(open(os.path.join(T, "guard", "i5b18-guard-01.json"), encoding="utf-8"))
out = ["İ5B-FIX-18 guard removal — one guard undone at a time in ONE work copy outside the repository, restored before every",
       "mutation (rsync from the İ5B-FIX-14 base, the two frozen files laid over it, checked by SHA-256); run by the AI session.",
       "Each mutation: the file still parses; the named test fails without a crash; the control test",
       "(test_a_dead_link_is_still_a_dead_link) passes. Run: drivers/harness18.py.txt — 6 new guards (G-Q1 … G-Q4, G-Q6, G-Q7)",
       "and the 14 earlier guards whose path runs through the closing branch or an HTML block's start line. The other earlier",
       "guards' anchors are all still once in the final source and their code is untouched; they are not run again.", "",
       f"proven {sum(r.get('proven', False) for r in res)} of {len(res)}"]
for r in res:
    out += [f"  {r['mutation']}", f"      tests: {', '.join(t.split('.')[-1] for t in r['tests'])} · named: {'RED' if r['red'] else 'green'} — {r['summary']}"
            f" · control: {r['control_summary']} · proven: {r['proven']}"]
fuzz = open(os.path.join(T, "same-end-fuzz.txt"), encoding="utf-8").read().strip()
out += ["", "Not a guard: 'the same end as HTML_TAG'. A paragraph closing tag counts only when CLOSING_TAG matches at the same",
        "position and reaches the same end as HTML_TAG (the agreed condition). No input was found where a CLOSING_TAG match",
        "ends elsewhere than HTML_TAG's, so undoing that half of the condition showed no effect; it is kept as a defensive",
        "check, with no guard that proves it. The search below is finite support, not a proof for every input",
        "(drivers/same-end-fuzz.py.txt, a seeded random search over closing-tag strings):", "  " + fuzz]
write("guard-removal.txt", out)

real = open(os.path.join(T, "real-data.txt"), encoding="utf-8").read().splitlines()
out = ["İ5B-FIX-18 — what was run on this branch's two frozen files, and what was not.", "",
       "sources (SHA-256):"] + [f"  {p:50s}{h}" for p, h in freeze.items()]
out += ["", "As agreed, the 54 budget tests and --check are not recorded here as a separate bound run: the one recorded run is",
        "the combined candidate's, after the commit (C17). During the work the suite passed 54 / 54 and --check reported 9",
        "adapters; that is not a recorded run.", "",
        "The real files, C16's reader against this one:"] + ["  " + l for l in real]
write("checks.txt", out)

before = json.load(open(os.path.join(S, "probe-i5b18-before-01.json"), encoding="utf-8"))
after = json.load(open(os.path.join(S, "probe-i5b18-final-01.json"), encoding="utf-8"))
out = ["İ5B-FIX-18 real CLI — 15 cases (the normal file and this package's negatives and positives), write and --check, each in",
       "ONE work copy restored before every case; run by the AI session. Driver: drivers/probe_i5b18.py.txt — the tool laid over",
       "İ5B-FIX-14's frozen base: C16's (before) and the frozen one (after). One AGENTS.md case passes 22000 bytes and prints",
       "WARN. NBSP follows the normative CommonMark model (it is not CommonMark whitespace); no claim is made about every",
       "renderer.", "", f"{'case':96s} {'expected':9s} {'before':7s} after"]
for b, a in zip(before["results"], after["results"]):
    assert b["case"] == a["case"]
    out.append(f"{b['case'][:96]:96s} {b['expected']:9s} {b['write_check']:7s} {a['write_check']}")
tools = lambda d: sorted({r["tool_sha256"] for r in d["results"]})
out += ["", f"before: tool {tools(before)}, {sum(r['ok'] for r in before['results'])} of {len(before['results'])} as expected",
        f"after:  tool {tools(after)}, {sum(r['ok'] for r in after['results'])} of {len(after['results'])} as expected",
        f"base unchanged before / after: {before['base_tree_sha256_before'] == before['base_tree_sha256_after']} / {after['base_tree_sha256_before'] == after['base_tree_sha256_after']}",
        f"results: before {sha(os.path.join(S, 'probe-i5b18-before-01.json'))}, after {sha(os.path.join(S, 'probe-i5b18-final-01.json'))} (kept outside the repository)"]
write("real-cli.txt", out)
for src, dst in (("i5b18/guard/harness18.py", "harness18.py.txt"), ("probe_i5b18.py", "probe_i5b18.py.txt"),
                 ("i5b18/same-end-fuzz.py", "same-end-fuzz.py.txt"), ("make_evidence19.py", "make_evidence19.py.txt")):
    shutil.copyfile(os.path.join(S, src), os.path.join(E, "drivers", dst))
print("written")
