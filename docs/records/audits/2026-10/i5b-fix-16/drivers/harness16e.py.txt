CONTROL = B + "test_a_dead_link_is_still_a_dead_link"
sha = lambda p: hashlib.sha256(open(p, "rb").read()).hexdigest()


def restore():
    subprocess.run(["rsync", "-a", "--delete", "--checksum", BASE + "/", W + "/"], check=True)
    shutil.copyfile(TOOL, os.path.join(W, F)); shutil.copyfile(TEST, os.path.join(W, "scripts/tests/test_sync_adapters_budget.py"))
    for p, h in FROZEN.items():
        assert sha(os.path.join(W, p)) == h, p


def run(names):
    try:
        p = subprocess.run([sys.executable, "-B", "-m", "unittest", *names], cwd=W, capture_output=True, text=True, timeout=600)
    except subprocess.TimeoutExpired:
        return "timeout", "the test did not finish in time"
    crashed = "Traceback" in p.stderr and "AssertionError" not in p.stderr
    return (99 if crashed else p.returncode), (p.stderr.strip().splitlines() or [""])[-1]


os.makedirs(W, exist_ok=True)
results = []
for label, old, new, tests in M:
    restore()
    f = os.path.join(W, F)
    text = open(f, encoding="utf-8").read()
    n = text.count(old)
    if n != 1:
        results.append({"mutation": label, "applied": False, "anchor_count": n}); print(label, "NOT APPLIED", n, flush=True); continue
    open(f, "w", encoding="utf-8").write(text.replace(old, new))
    parsed = subprocess.run([sys.executable, "-B", "-c", "import ast,sys; ast.parse(open(sys.argv[1]).read())", f]).returncode == 0
    t0 = time.time()
    code, tail = run(tests)
    ccode, ctail = run([CONTROL])
    proven = parsed and code not in (0, 99, "timeout") and ccode == 0
    results.append({"mutation": label, "applied": True, "file": F, "tests": tests, "exit": code, "red": code != 0, "summary": tail,
                    "parsed": parsed, "control": CONTROL, "control_exit": ccode, "control_summary": ctail, "proven": proven,
                    "seconds": round(time.time() - t0)})
    print(label, "PROVEN" if proven else "NOT PROVEN", "| named:", tail, "| control:", ctail, flush=True)
restore()
json.dump(results, open(OUT, "w"), ensure_ascii=False, indent=1)
print("proven", sum(r.get("proven", False) for r in results), "of", len(results), "· the copy restored to the frozen sources")
