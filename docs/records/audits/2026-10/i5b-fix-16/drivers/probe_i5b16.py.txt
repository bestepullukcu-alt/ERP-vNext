import difflib, hashlib
def tree_id(root):                                   # one SHA-256 over every file's path and bytes
    h = hashlib.sha256()
    for d, dirs, fs in os.walk(root):
        dirs.sort()
        for f in sorted(fs):
            p = os.path.join(d, f); h.update(os.path.relpath(p, root).encode() + b"\0" + open(p, "rb").read() + b"\0")
    return h.hexdigest()
TOOL = sys.argv[4] if len(sys.argv) > 4 else None   # optional: the frozen sync_adapters.py laid over the base each round
def restore():
    subprocess.run(["rsync", "-a", "--delete", "--checksum", SRC + "/", W + "/"], check=True)
    if TOOL:
        shutil.copyfile(TOOL, os.path.join(W, ".antigravity/scripts/sync_adapters.py"))
def sha(b):
    return hashlib.sha256(b).hexdigest()
NAME = sys.argv[3]                                   # the result name: <NAME>.json and the <NAME>/ folder
OUT = os.path.join(os.path.dirname(WORK), NAME); os.makedirs(OUT)
W = os.path.join(WORK, "one"); os.makedirs(W, exist_ok=True)
base_before = tree_id(SRC); tool = ".antigravity/scripts/sync_adapters.py"
restore(); src_before = tree_id(W)                   # the base with the tool laid over it
res = []
for n, (label, files, want) in enumerate(cases):
    restore()
    assert tree_id(W) == src_before, "the single copy is not the source before case %d" % n
    tool_sha = sha(open(os.path.join(W, tool), "rb").read())
    inputs = {}
    for p, t in files.items():
        open(os.path.join(W, p), "w", encoding="utf-8").write(t)
        old = open(os.path.join(SRC, p), encoding="utf-8").read()
        d = "".join(difflib.unified_diff(old.splitlines(True), t.splitlines(True), "a/" + p, "b/" + p))
        open(os.path.join(OUT, f"c{n:02d}.input.diff"), "a", encoding="utf-8").write(d)
        inputs[p] = {"sha256": sha(t.encode()), "bytes": len(t.encode())}
    run = lambda *x: subprocess.run([sys.executable, "-B", tool, *x], cwd=W, capture_output=True, text=True, timeout=120)
    a, b = run(), run("--check"); got = f"{a.returncode}/{b.returncode}"
    open(os.path.join(OUT, f"c{n:02d}.output.txt"), "w", encoding="utf-8").write(
        f"case: {label}\n### write exit {a.returncode}\n{a.stdout}{a.stderr}### check exit {b.returncode}\n{b.stdout}{b.stderr}")
    out = (a.stdout + a.stderr).splitlines()
    ok = (got == "0/0") == (want == "0/0")
    res.append({"case": label, "n": n, "write_check": got, "expected": want, "ok": ok, "tool_sha256": tool_sha, "inputs": inputs,
                "warn": any("WARN" in l for l in out),
                "problems": [l[:240] for l in out if l.startswith(("HEADING", "SECTION", "ANCHOR", "CORE"))][:3]})
    print(f"{'OK ' if ok else 'GAP'} {label}: {got} (expected {want}){' WARN' if res[-1]['warn'] else ''} · {res[-1]['problems'][:1]}")
restore()
summary = {"base_tree_sha256_before": base_before, "base_tree_sha256_after": tree_id(SRC), "tool_override": bool(TOOL),
           "source_tree_sha256_before": src_before, "source_tree_sha256_after": (restore(), tree_id(W))[1], "single_copy_after_restore": tree_id(W),
           "cases": len(res), "all_as_expected": all(r["ok"] for r in res), "results": res}
