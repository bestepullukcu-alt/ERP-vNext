"""İ5B-FIX-11: the generated evidence texts from the raw outputs (no local paths).
Usage: make_evidence10.py <scratch ctfix09> <İ5B-FIX-11 evidence dir>"""
import hashlib, json, os, re, shutil, sys
S, I = sys.argv[1:3]
os.makedirs(os.path.join(I, "drivers"), exist_ok=True)


def sha(p):
    return hashlib.sha256(open(os.path.join(S, p), "rb").read()).hexdigest()


def write(name, lines):
    open(os.path.join(I, name), "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def rows_of(text):
    started = re.findall(r"(?:^|(?<=\.\.\. ))(test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\) \.\.\.", text, re.M)   # subtest lines run together
    fails = re.findall(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\)(?: (\(.*\)))?$", text, re.M)
    bad = {}
    for res, name, mod, cls, _ in fails:
        bad.setdefault(f"{mod}.{cls}.{name}", res)
    return ([(bad.get(f"{mod}.{cls}.{name}", "ok"), f"{mod}.{cls}.{name}") for name, mod, cls in started],
            [(res, name, sub) for res, name, mod, cls, sub in fails if sub])

out += [f"  {res:5s} {tid}" for res, tid in rows]

out += ["", "The two new tests fail on the base source: every shape case (the hidden titles, the declared false refusals and the",
        "real-shape breaks) passes there (10), every reference-image case passes (5), and the escaped link tail 'a\\](b)' is",

        "had the same results; the final files differ from it only by narrower wording (a docstring, a test label) and one",

       "is not counted for these sources.", ""]
