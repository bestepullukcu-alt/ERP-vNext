// WP-0177-R2 sabotage tool: node sabotage-r2.js <repoRoot> apply|restore <id>
// apply: backs up the target file(s) to <file>.sabotage-bak and replaces exact text (must match once); restore: puts the backups back
// and stamps the mtime (a restored file must be newer than the sabotaged build, or MSBuild keeps the sabotaged assembly).
const fs = require('fs');
const path = require('path');
const [root, mode, id] = process.argv.slice(2);
const src = 'services/Diten.InventoryService/src/';
const engine = src + 'Diten.InventoryService.Application/Features/Recall/Trace/RecallTraceEngine.cs';
const rules = src + 'Diten.InventoryService.Domain/Recall/Trace/TraceRules.cs';
const reader = src + 'Diten.InventoryService.Infrastructure/Recall/ShipmentBundleHttpReader.cs';
const S = {
  // (a) no visited set: a loop is walked again until the depth limit
  a: [[engine, 'var seen = visited.TryGetValue(other, out var seenAt);', 'var seen = false; var seenAt = 0; _ = visited;']],
  // (b) "Takip dışı" counted as a gap
  b: [[engine, 'else if (controlled == false)', 'else if (controlled == false && false)']],
  // (c) totals across units
  c: [[rules, '.GroupBy(n => (n.SkuId!.Value, n.BaseUomId!))', '.GroupBy(n => (n.SkuId!.Value, string.Empty))']],
  // (d) shipment reads cached
  d: [[reader, 'public const int MaxParallel = 8;',
       'public const int MaxParallel = 8;\n    private static readonly System.Collections.Concurrent.ConcurrentDictionary<string, TraceShipmentInfo> Cache = new(StringComparer.Ordinal);'],
      [reader, 'return (id, await ReadOneAsync(id, cancellationToken));',
       'return (id, Cache.TryGetValue(id, out var hit) ? hit : Cache[id] = await ReadOneAsync(id, cancellationToken));']],
  // (e) node order changed (levels descending)
  e: [[rules, '            .ThenBy(n => n.Level)\n', '            .ThenByDescending(n => n.Level)\n']],
  // (f) the other company's lot shown on OTHER_COMPANY_ACTIVITY
  f: [[engine, '                GapCode = TraceGapCode.OTHER_COMPANY_ACTIVITY,\n',
       '                GapCode = TraceGapCode.OTHER_COMPANY_ACTIVITY,\n                LotId = root.Id.ToString("D"), LotNumber = root.LotNumber,\n']],
  // (g) no LIMIT_REACHED gap
  g: [[engine, '        if (run.Truncated)\n', '        if (false)\n']],
  // (h) approval does not refuse the changed live trace
  h: [[src + 'Diten.InventoryService.Application/Features/Recall/Handlers/CaseCommandHandlers.cs',
       '        if (drift is not null) return RecallScopeDrift.Refusal<RecallCaseDetailDto>(drift);\n', '        _ = drift;\n']],
  // (i) a shipment 401 stops the trace
  i: [[reader, 'case HttpStatusCode.Unauthorized or HttpStatusCode.Forbidden:\n                    return new TraceShipmentInfo(ShipmentReadOutcome.Forbidden, null, null, null);',
       'case HttpStatusCode.Unauthorized or HttpStatusCode.Forbidden:\n                    throw new UnauthorizedAccessException("shipment read refused");']]
};
const edits = S[id];
if (!edits) throw new Error('unknown sabotage ' + id);
const touched = new Set();
for (const [rel, from, to] of edits) {
  const file = path.join(root, rel);
  const bak = file + '.sabotage-bak';
  if (mode === 'apply') {
    if (!fs.existsSync(bak)) fs.copyFileSync(file, bak);
    let s = fs.readFileSync(file, 'utf8').replace(/\r\n/g, '\n');
    const n = s.split(from).length - 1;
    if (n !== 1) throw new Error(`sabotage ${id}: ${rel} anchor matched ${n} times`);
    fs.writeFileSync(file, s.replace(from, () => to));
    const now = new Date(); fs.utimesSync(file, now, now);
  } else if (!touched.has(file) && fs.existsSync(bak)) {
    touched.add(file);
    fs.copyFileSync(bak, file);
    fs.unlinkSync(bak);
    const now = new Date(); fs.utimesSync(file, now, now);
  }
}
console.log(`${mode} ${id} ok`);
