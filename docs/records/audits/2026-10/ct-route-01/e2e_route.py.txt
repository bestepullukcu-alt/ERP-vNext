the worktree's HEAD) — used to start from the merged tree's single-parent fixture base.
Every commit is synthetic and lives only in the scratch clone; nothing is written to the worktree."""

def git(*a: str, cwd: Path = C) -> str:
    return subprocess.run(["git", "-C", str(cwd), *a], capture_output=True, text=True, env=ENV, check=True).stdout.strip()


def write(rel: str, text: str):
    p = C / rel
    p.parent.mkdir(parents=True, exist_ok=True)  # inside the scratch clone only
    p.write_text(text, encoding="utf-8")


def commit(msg: str, files: dict[str, str]) -> str:
    for rel, text in files.items():
        write(rel, text)
    git("add", "--", *files)
    git("commit", "-q", "-m", msg)
    return git("rev-parse", "HEAD")


def variant(base: str, msg: str, files: dict[str, str]) -> str:
    git("checkout", "-q", "--detach", base)
    sha = commit(msg, files)
    git("checkout", "-q", "--detach", base)
    return sha


def main_at(sha: str | None):
    if sha is None:
        git("update-ref", "-d", "refs/remotes/origin/main")
    else:
        git("update-ref", "refs/remotes/origin/main", sha)


def run(label: str, *args: str, tool_dir: Path = C) -> tuple[int, str]:
    done = subprocess.run([sys.executable, "-B", str(tool_dir / "scripts/ct_route.py"), *args], cwd=tool_dir, capture_output=True,
                          text=True, env=ENV)
    out = done.stdout + (("\n[stderr]\n" + done.stderr) if done.stderr.strip() else "")
    log.append(f"## {label} — exit {done.returncode}")
    log.extend("   " + l[:220] for l in out.splitlines()
               if l.startswith(("sonuç", "aday", "yürürlük", "önce oku", "kişi", "contacts", "  Kime", "iş:", "[stderr]", "Traceback")))
    return done.returncode, out


def expect(label: str, code: int, out: str, want: int, *contains: str, absent: tuple[str, ...] = ()):
    ok = code == want and all(c in out for c in contains) and not any(a in out for a in absent) and "Traceback" not in out
    log.append(f"   {'PASS' if ok else 'FAIL'} expected: exit {want}" + (f", contains {list(contains)}" if contains else "")
               + (f", absent {list(absent)}" if absent else ""))
    if not ok:
        failures.append(label)


def set_cell(text: str, qid: str, column: str, value: str) -> str:
    lines = text.rstrip("\n").split("\n")
    head = lines[0].split("\t")
    out = [lines[0]]
    for l in lines[1:]:
        cells = l.split("\t")
        if cells[0] == qid:
            cells[head.index(column)] = value
        out.append("\t".join(cells))
    return "\n".join(out) + "\n"
