lines = []
for n, (name, old, new) in enumerate(G, 1):
    c = src.count(old)
    if c != 1:
        lines.append(f"[ct_route] {n:03d} HARNESS: anchor for '{name}' found {c} times")
        continue
    path = OUT / f"sab_{n:03d}.py"
    path.write_text(src.replace(old, new), encoding="utf-8")  # scratch copy written by this harness
    spec = importlib.util.spec_from_file_location(f"sab{n}", path); m = importlib.util.module_from_spec(spec)
    buf = io.StringIO()
    try:
        spec.loader.exec_module(m)
        m.ROOT = W
        with contextlib.redirect_stdout(buf):
            m.self_test()
        last = buf.getvalue().splitlines()[-1]
    except Exception as e:
        last = f"self-test crashed ({type(e).__name__})"
    fails = [l[5:] for l in buf.getvalue().splitlines() if l.startswith("FAIL ")]
    lines.append(f"[ct_route] {n:03d} guard removed: {name} -> {last}; red: {' | '.join(f[:100] for f in fails) or 'NONE'}")
(OUT / "sabotage.txt").write_text("\n".join(lines) + "\n", encoding="utf-8")
print("\n".join(l[:240] for l in lines if "red: NONE" in l or "crashed" in l or "HARNESS" in l) or "(every guard turned a named test red)")
print(f"guards {len(G)} · no red {sum('red: NONE' in l for l in lines)} · crashed {sum('crashed' in l for l in lines)} · harness {sum('HARNESS' in l for l in lines)}")
