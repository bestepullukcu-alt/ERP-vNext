results = []
for i, (label, path, old, new, tests) in enumerate(CT if LINE == "ct" else I5B):
    d = os.path.join(WORK, f"g{i:02d}")
    shutil.copytree(SRC, d)
    f = os.path.join(d, path)
    text = open(f, encoding="utf-8").read()
    n = text.count(old)
    if n != 1:
        results.append({"mutation": label, "applied": False, "anchor_count": n}); print(label, "NOT APPLIED", n, flush=True); continue
    open(f, "w", encoding="utf-8").write(text.replace(old, new))
    compiled = subprocess.run([sys.executable, "-B", "-c", "import ast,sys; ast.parse(open(sys.argv[1]).read())", f]).returncode == 0
    t0 = time.time()

    def run(names):
        selftests = [n[len("self-test:"):] for n in names if n.startswith("self-test:")]
        units = [n for n in names if not n.startswith("self-test:")]
        codes, tails = [], []
        try:
            for s in selftests:   # 'script::case': red only when that named case is reported FAIL, without a crash
                script, _, case = s.partition("::")
                p = subprocess.run([sys.executable, "-B", script, "--self-test"], cwd=d, capture_output=True, text=True, timeout=240)
                crashed = "Traceback" in p.stdout + p.stderr
                named = any(l.startswith("FAIL " + case) for l in p.stdout.splitlines()) if case else p.returncode != 0
                codes.append(1 if named and not crashed else (99 if crashed else 0))
                tails.append(("crash: " + (p.stderr.strip().splitlines() or [""])[-1][:120]) if crashed else
                             (f"FAIL {case[:60]}" if named else (p.stdout.strip().splitlines() or [""])[-1]))
            if units:
                p = subprocess.run([sys.executable, "-B", "-m", "unittest", *units], cwd=d, capture_output=True, text=True, timeout=600)
                codes.append(p.returncode); tails.append((p.stderr.strip().splitlines() or [""])[-1])
        except subprocess.TimeoutExpired:
            return "timeout", "the test did not finish in time (a hang is a failure)"
        return (max(codes) if codes else 0), " · ".join(tails)
    code, tail = run(tests)
    ccode, ctail = run([CONTROL])   # the intact path still works in the same copy: the red is the guard, not a crash
    proven = compiled and code not in (0, 99) and ccode == 0   # 99: a crash is not the named test failing
    results.append({"mutation": label, "applied": True, "file": path, "tests": tests, "exit": code, "red": code != 0,
                    "summary": tail, "parsed": compiled, "control": CONTROL, "control_exit": ccode, "control_summary": ctail,
                    "proven": proven, "seconds": round(time.time() - t0)})
    print(label, "PROVEN" if proven else "NOT PROVEN", "| named:", tail, "| control:", ctail, flush=True)
json.dump(results, open(OUT, "w"), ensure_ascii=False, indent=1)
print("proven", sum(r.get("proven", False) for r in results), "of", len(results))
