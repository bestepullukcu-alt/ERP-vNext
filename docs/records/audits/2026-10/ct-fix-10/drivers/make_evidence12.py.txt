"""CT-FIX-10: the generated evidence texts from the raw outputs (no local paths).
Usage: make_evidence12.py <scratch ctfix10> <CT-FIX-10 evidence dir>"""
import hashlib, json, os, re, shutil, sys
S, E = sys.argv[1:3]
os.makedirs(os.path.join(E, "drivers"), exist_ok=True)


def sha(p):
    return hashlib.sha256(open(os.path.join(S, p), "rb").read()).hexdigest()


def write(name, lines):
    open(os.path.join(E, name), "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def rows_of(text):
    started = re.findall(r"(?:^|(?<=\.\.\. ))(test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\) \.\.\.", text, re.M)   # subtest lines run together
    fails = re.findall(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\)(?: (\(.*\)))?$", text, re.M)
    bad = {}
    for res, name, mod, cls, _ in fails:
        bad.setdefault(f"{mod}.{cls}.{name}", res)
    return ([(bad.get(f"{mod}.{cls}.{name}", "ok"), f"{mod}.{cls}.{name}") for name, mod, cls in started],
            [(res, name, sub) for res, name, mod, cls, sub in fails if sub])

text = open(os.path.join(S, "red-ct10-final.txt"), encoding="utf-8").read()

out += [f"  {res:5s} {tid}" for res, tid in rows]
out += ["", "failed subtests:"] + [f"  {res:5s} {name} {sub}" for res, name, sub in subs]
out += ["", "On the base sources, under a broken registry: a sound product pack gets a false 'paket … bu işin modülüne (yok) ait",
        "değil' and no 'ölçülemedi' (1); a pack whose künye id and file name disagree is already refused by its own identity",
        "check there, and only the 'ölçülemedi' line is missing (1). writers crashes with a traceback (exit 1) when a local ref,",
        "the base or only the merge base holds a queue that is not UTF-8 (3). The controls in the same tests pass there: an intact queue with a sound",
        "pack (0), a WP not in the queue (still refused as a module mismatch), another module's pack (still refused), and",
        "writers on the normal fixture (0). Controls.* pass.", "",
        "Earlier, not replaced: the tests first ran red before the code was written (4 failures). Two test mistakes found",
        "afterwards were corrected in the test, not in the code: the künye / file-name case expected no 'ölçülemedi' line (the",
        "gate rightly writes both), and the merge-base case kept an earlier subtest's branch (each subtest now drops its own)."]
write("red-first.txt", out)

res = json.load(open(os.path.join(S, "guard/ct-guard-01.json"), encoding="utf-8"))
same = all(hashlib.sha256(open(os.path.join(S, "guard/ct-src", p), "rb").read()).hexdigest() == h for p, h in freeze.items())

       f"sources byte for byte: {same}. Each mutation: the file still parses; the named test fails; the control test",
       "(Controls.test_consumers_on_the_intact_fixture_exit_0) passes.", ""]
for r in res:
    out += [r["mutation"] + f"  ({r['file']})", f"    tests: {', '.join(t.split('.')[-1] for t in r['tests'])}",
            f"    applied: {r['applied']} · parses: {r['parsed']} · named: {'RED' if r['red'] else 'green'} — {r['summary']}"
            f" · control: {r['control_summary']} · proven: {r['proven']}"]
