def red_first(raw, title, base, note, prefix):
    text = open(raw, encoding="utf-8").read()
    ran = re.search(r"^Ran (\d+) tests", text, re.M).group(1)
    per = collections.Counter()
    for m in re.finditer(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.\w+\.(\w+)\)", text, re.M):
        per[f"{m.group(3)}.{m.group(2)}" + (" (error)" if m.group(1) == "ERROR" else "")] += 1
    out = [title, f"base code {base} (exported with git archive outside the repository; this package's test file copied in).",
           note, "", f"ran: {ran} tests · failed subtests: {sum(per.values())} in {len(per)} test methods", ""]
    out += [f"  {n:4d}  {k.replace(prefix, '')}" for k, n in sorted(per.items())]
    return "\n".join(out) + "\n"

def guards(js, title, line_note):
    res = json.load(open(js, encoding="utf-8"))
    out = [title, line_note, "Driver: drivers/harness2.py.txt. Each mutation: the file still parses; the named tests fail (a hang",
           "past 240 s counts as a failure); the control test passes in the same copy (the red is the guard, not a crash).", ""]
    for r in res:
        out.append(r["mutation"])
        out.append(f"    file: {r['file']} · tests: {', '.join(r['tests'])}")
        out.append(f"    named: {'RED' if r['red'] else 'green'} — {r['summary']} · control {r['control']}: {r['control_summary']}"
                   f" · parsed: {r['parsed']} · proven: {r['proven']}")
    out += ["", f"proven: {sum(r['proven'] for r in res)} of {len(res)}"]
    return "\n".join(out) + "\n"
