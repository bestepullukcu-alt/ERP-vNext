"""Candidate 4 (scratch, no ref): candidate 3 (0609bc588) + the CT-FIX-05 files (worktree on chore/ct/ct-fix-05, base
f23455d27) + the İ5B-FIX-06 files (worktree on chore/agents/i5b-fix-06, base a7cb60992). The overlay is clean only when
each branch's base blob is candidate 3's blob and a new file is absent there. Usage: build.py <fixture> <ct wt> <i5b wt> <out dir>"""
import subprocess, sys, os, json, hashlib
A, CTW, IW, D = sys.argv[1:5]
BASE = "0609bc5882380580201e21a89fcbc2302cff884d"
LINES = {
    CTW: ("f23455d278ee65b4e25ddb892bad3567c3ce7e4c",
          ["scripts/acceptance_record.py", "scripts/ct_queue.py", "scripts/ct_route.py", "scripts/known_failures.py",
           "scripts/prompt_gate.py", "scripts/status_report_v2.py", "scripts/run_record.py",
           "scripts/tests/test_ct_cli_regression.py", "docs/guides/operations/control-tower-sop.md",
           "docs/guides/operations/control-tower-operating-card.md", ".antigravity/rules/status-reporting-and-evidence.md"]),
    IW: ("a7cb60992fbf355e6bcf947a3abe4f778bcefbc1",
         [".antigravity/scripts/sync_adapters.py", "AGENTS.md", ".claude/rules/agents-core.md",
          "scripts/tests/test_sync_adapters_budget.py"]),
}
env = dict(os.environ, GIT_OPTIONAL_LOCKS="0", GIT_INDEX_FILE=os.path.join(D, "index"),
           GIT_AUTHOR_NAME="CT-FIX-05 candidate", GIT_AUTHOR_EMAIL="candidate@example.invalid",
           GIT_COMMITTER_NAME="CT-FIX-05 candidate", GIT_COMMITTER_EMAIL="candidate@example.invalid",
           GIT_AUTHOR_DATE="2026-10-17T21:00:00Z", GIT_COMMITTER_DATE="2026-10-17T21:00:00Z")
WENV = {k: v for k, v in env.items() if k != "GIT_INDEX_FILE"}   # the worktrees are read with their own index
def git(*a, cwd=A, ok=False):
    p = subprocess.run(["git", *a], cwd=cwd, capture_output=True, text=True, env=env if cwd == A else WENV)
    if p.returncode and not ok: raise SystemExit(f"git {a}: {p.stderr}")
    return p.stdout.strip() if not p.returncode else None
for W, (WBASE, files) in LINES.items():
    # every changed or new file of the worktree is in the list, nothing else (the overlay is the branch's whole change)
    changed = sorted(set(git("diff", "--name-only", WBASE, cwd=W).splitlines())
                     | set(git("ls-files", "--others", "--exclude-standard", cwd=W).splitlines()))
    if changed != sorted(files): raise SystemExit(f"stop: {W} changes {changed} != list {sorted(files)}")
    for f in files:
        b1 = git("rev-parse", f"{WBASE}:{f}", cwd=W, ok=True)
        b2 = git("rev-parse", f"{BASE}:{f}", ok=True)
        print("base blob", "same" if b1 == b2 else "DIFFERENT", "(new)" if b1 is None else "", f)
        if b1 != b2: raise SystemExit("stop: not a clean overlay")
git("read-tree", BASE)
manifest = {}
for W, (WBASE, files) in LINES.items():
    for f in files:
        blob = git("hash-object", "-w", "--path", f, os.path.join(W, f))
        mode = "100755" if os.access(os.path.join(W, f), os.X_OK) else "100644"
        git("update-index", "--add", "--cacheinfo", f"{mode},{blob},{f}")
        manifest[f] = {"line": "CT-FIX-05" if W == CTW else "İ5B-FIX-06", "base": WBASE, "mode": mode, "blob": blob,
                       "sha256": hashlib.sha256(open(os.path.join(W, f), "rb").read()).hexdigest()}
tree = git("write-tree")
commit = git("commit-tree", tree, "-p", BASE, "-m",
             "CT-FIX-05 / İ5B-FIX-06 candidate 4 (scratch, no ref): candidate 3 (0609bc588) + the v6 audit fixes F1, F2(b), "
             "F3 run record, F4, F5, F6, C1, C2, E1 and the four-part module-request answer, with their tests and documents")
names = git("diff", "--name-status", "--no-renames", BASE, commit).splitlines()
json.dump({"base": BASE, "tree": tree, "commit": commit, "diff_vs_base": names, "files": manifest},
          open(os.path.join(D, "candidate.json"), "w"), indent=1, ensure_ascii=False)
print(commit, tree); print("\n".join(names))
