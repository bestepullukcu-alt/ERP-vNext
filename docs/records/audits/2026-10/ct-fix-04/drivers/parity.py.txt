"""Real-data parity: the same read-only commands in one repository; stdout/stderr/exit recorded verbatim."""
import json, subprocess, sys, os
REPO, OUT = sys.argv[1:3]
CMDS = [
 ["scripts/ct_queue.py", "validate"],
 ["scripts/ct_queue.py", "summary", "--queue", "sce-mvp6"],
 ["scripts/ct_queue.py", "guard", "--queue", "sce-mvp6"],
 ["scripts/ct_queue.py", "writers", "--queue", "sce-mvp6"],
 ["scripts/ct_queue.py", "require", "Q149", "--queue", "sce-mvp6"],
 ["scripts/ct_queue.py", "require", "Q149", "--queue", "sce-mvp6", "--in-force"],
 ["scripts/ct_queue.py", "require", "Q01", "--queue", "sce-mvp6"],
 ["scripts/ct_route.py", "--kind", "delivery", "--module", "MOD-0185"],
 ["scripts/ct_route.py", "--kind", "delivery", "--module", "MOD-0251"],
 ["scripts/ct_route.py", "--kind", "contract", "--module", "MOD-0251"],
 ["scripts/ct_route.py", "--kind", "contract", "--module", "MOD-0173"],
 ["scripts/ct_route.py", "--kind", "delivery", "--wp", "Q149", "--queue", "sce-mvp6"],
 ["scripts/status_report.py", "--mode", "v2", "--queue", "sce-mvp6"],
 [".antigravity/scripts/sync_adapters.py", "--check"],
]
env = dict(os.environ, GIT_OPTIONAL_LOCKS="0", PYTHONDONTWRITEBYTECODE="1")
head = subprocess.run(["git", "rev-parse", "HEAD"], cwd=REPO, capture_output=True, text=True, env=env).stdout.strip()
res = {"head": head, "runs": []}
for c in CMDS:
    p = subprocess.run([sys.executable, "-B", *c], cwd=REPO, capture_output=True, text=True, env=env, timeout=600)
    res["runs"].append({"cmd": " ".join(c), "exit": p.returncode, "stdout": p.stdout, "stderr": p.stderr})
    print(p.returncode, " ".join(c), flush=True)
dirty = subprocess.run(["git", "status", "--porcelain=v1"], cwd=REPO, capture_output=True, text=True, env=env).stdout
res["dirty_after"] = dirty
json.dump(res, open(OUT, "w"), ensure_ascii=False, indent=1)
print("dirty after:", len(dirty.splitlines()))
