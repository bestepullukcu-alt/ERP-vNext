ok_all = True
for sid, rel, old, new, test, name in S:
    path = FIX / rel
    raw = path.read_bytes()
    text = raw.decode("utf-8")
    if text.count(old) != 1:
        print(f"VACUOUS {sid}: anchor found {text.count(old)} times"); ok_all = False; continue
    path.write_bytes(text.replace(old, new).encode("utf-8"))
    try:
        r = subprocess.run([sys.executable, "-B", test, "--self-test"], cwd=FIX, capture_output=True, text=True, env=ENV, timeout=900)
    finally:
        path.write_bytes(raw)
    restored = hashlib.sha256(path.read_bytes()).hexdigest() == hashlib.sha256(raw).hexdigest()
    fails = [l for l in r.stdout.splitlines() if l.startswith("FAIL")]
    named = [l for l in fails if name in l]
    red = r.returncode != 0 and bool(named)
    ok_all = ok_all and red and restored
    summary = next((l for l in r.stdout.splitlines() if l.startswith("self-test:")), r.stderr.strip()[-120:])
    print(f"{'RED ' if red else 'MISS'} {sid}\n      {test} exit {r.returncode}; {summary}; named FAIL: {named[0][5:110] if named else '-'}"
          f"; other FAILs {len(fails) - len(named)}; restored {restored}")
print(f"\n{sum(1 for _ in S)} sabotages; all named tests red and every file restored: {ok_all}")
