def git(*args, check=True):
    r = subprocess.run(["git", "-C", str(FIX), "-c", "commit.gpgsign=false", *args], capture_output=True, text=True, env=ENV)
    if check and r.returncode != 0:
        raise SystemExit(f"git {' '.join(args)}: {r.stderr}")
    return r.stdout.strip()


def put(rel, text):
    p = FIX / rel
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(text, encoding="utf-8")


def commit(msg, *paths):
    git("add", "--", *paths)
    git("commit", "-q", "--allow-empty", "-m", msg)
    return git("rev-parse", "HEAD")


def tool(*args):
    r = subprocess.run([sys.executable, "-B", *args], cwd=FIX, capture_output=True, text=True, env=ENV, timeout=600)
    return r.returncode, r.stdout, r.stderr


def first(text, pat):
    m = next((l for l in text.splitlines() if re.search(pat, l)), "")
    return m.strip()[:160]


def case(cid, expected, observed, red):
    results.append({"id": cid, "expected": expected, "observed": observed, "red": bool(red)})
    print(f"{'RED ' if red else 'ok  '} {cid}\n      expected: {expected}\n      observed: {observed}")

def ledger_text(at):
    return ("---\nformat: effort-ledger-v1\nqueue: ctf2\nbaseline_version: 1\nunit: saat\n"
            "weights: backend=0.5 frontend=0.3 test=0.2\napproval_record: " + APPROVAL + "@" + at + "\n---\n\n"
            "| ID | Backend | Frontend | Test |\n|---|---|---|---|\n| W1 | 8 | 4 | 4 |\n| W2 | 8 | 0 | 4 |\n| W3 | 2 | 2 | 2 |\n")


def content_sha():
    import hashlib
    t = ledger_text("0" * 40)
    return hashlib.sha256("\n".join(l for l in t.split("\n") if not l.startswith("approval_record:")).encode()).hexdigest()


def report_at(status):
    """A side commit chain on top of c2: an approval record with this status, then the ledger naming it. Returns S."""
    git("checkout", "-q", "--detach", c2)
    put(APPROVAL, f"---\nstatus: {status}\nledger: {LEDGER}\nbaseline_version: 1\nledger_content_sha256: {content_sha()}\n---\n\nonay\n")
    a = commit(f"fixture: effort approval ({status})", APPROVAL)
    put(LEDGER, ledger_text(a))
    return commit("fixture: effort ledger", LEDGER)


def report(ref):
    code, out, err = tool("scripts/status_report.py", "--mode", "v2", "--queue", "ctf2", "--ref", ref)
    m = re.search(r"```json status-report-v2\n(.*?)\n```", out, re.S)
    if not m:
        return code, None, first(out, r"sonuç|tamamlanamadı") or err.strip()[-160:]
    d = json.loads(m.group(1))
    return code, d, f"complete={d['complete']} percentages_valid={d['percentages_valid']} in_force={d['source']['in_force']}"

def main_ref(text_or_commit):
    """origin/main imitation: a commit on c1 (the queue file exists there) whose registry is the given text; or an
    existing commit."""
    if re.fullmatch(r"[0-9a-f]{40}", text_or_commit):
        git("update-ref", "refs/remotes/origin/main", text_or_commit)
        return
    git("checkout", "-q", "--detach", c1)
    put(REG, text_or_commit)
    m = commit("fixture: origin/main imitation", REG)
    git("update-ref", "refs/remotes/origin/main", m)


def at_head(sha):
    git("checkout", "-q", "--detach", "-f", sha)
