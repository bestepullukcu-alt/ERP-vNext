def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def write(path, lines):
    open(path, "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def rows_of(text):
    started = re.findall(r"(?:^|(?<=\.\.\. ))(test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\) \.\.\.", text, re.M)   # subtest lines run together
    fails = re.findall(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\)(?: (\(.*\)))?$", text, re.M)
    bad = {}
    for res, name, mod, cls, _ in fails:
        bad.setdefault(f"{mod}.{cls}.{name}", res)
    out = [(bad.get(f"{mod}.{cls}.{name}", "ok"), f"{mod}.{cls}.{name}") for name, mod, cls in started]
    return out, [(res, f"{mod}.{cls}.{name}", sub) for res, name, mod, cls, sub in fails if sub]


def red(src, title, expect):
    text = open(src, encoding="utf-8").read()
    rows, subs = rows_of(text)
    out = title + ["", re.search(r"^Ran .*$", text, re.M).group(0) + " · " + re.search(r"^(OK|FAILED).*$", text, re.M).group(0), ""]
    out += [f"  {res:5s} {tid}" for res, tid in rows]
    if subs:
        out += ["", "failed subtests:"] + [f"  {res:5s} {tid.split('.')[-1]} {sub}" for res, tid, sub in subs]
    return out + [""] + expect

def guard(js, title):
    res = json.load(open(js, encoding="utf-8"))
    out = list(title) + [""]
    for r in res:
        out += [r["mutation"] + f"  ({r['file']})", f"    tests: {', '.join(t if t.startswith('self-test:') else t.split('.')[-1] for t in r['tests'])}",
                f"    applied: {r['applied']} · parses: {r['parsed']} · named: {'RED' if r['red'] else 'green'} — {r['summary']}"
                f" · control: {r['control_summary']} · proven: {r['proven']}"]
    return out + ["", f"proven: {sum(r['proven'] for r in res)} of {len(res)}"]


def copy_equal(src, frz):
    return all(sha(os.path.join(S, "guard", src, p)) == h for p, h in frz.items())

def run_lines(log, pre, post):
    text = open(os.path.join(S, log), encoding="utf-8").read()
    res = [l for l in text.splitlines() if l.startswith(("### exit", "Ran ", "OK", "FAILED", "start ", "end ")) or "self-test:" in l
           or l.startswith("OK ") or l.startswith("Python")]
    warn = len(re.findall(r"ResourceWarning: unclosed", text))
    a, b = open(os.path.join(S, pre), encoding="utf-8").read().splitlines(), open(os.path.join(S, post), encoding="utf-8").read().splitlines()
    added, removed = [l for l in b if l not in a], [l for l in a if l not in b]
    same = "True" if not added and not removed else (
        f"the same HEAD, branch, modified files and SHA-256; status differs only by {', '.join(x.strip() for x in added + removed)}"
        if all(x.startswith("?? docs/records/decisions/") for x in added) and not removed else f"False ({added} / {removed})")
    return res, warn, same
