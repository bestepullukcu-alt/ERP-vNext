"""CT-FIX-11: the generated evidence texts from the raw outputs (no local paths).
Usage: make_evidence15.py <scratch ctfix10/ctfix11> <CT-FIX-11 evidence dir>"""
import hashlib, json, os, re, shutil, sys
S, E = sys.argv[1:3]
os.makedirs(os.path.join(E, "drivers"), exist_ok=True)


def sha(p):
    return hashlib.sha256(open(os.path.join(S, p), "rb").read()).hexdigest()


def write(name, lines):
    open(os.path.join(E, name), "w", encoding="utf-8").write("\n".join(l.rstrip() for l in lines) + "\n")


def rows_of(text):
    started = re.findall(r"(?:^|(?<=\.\.\. ))(test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\) \.\.\.", text, re.M)   # subtest lines run together
    fails = re.findall(r"^(FAIL|ERROR): (test_\w+) \(scripts\.tests\.(\w+)\.(\w+)\)(?: (\(.*\)))?$", text, re.M)
    bad = {}
    for res, name, mod, cls, _ in fails:
        bad.setdefault(f"{mod}.{cls}.{name}", res)
    return ([(bad.get(f"{mod}.{cls}.{name}", "ok"), f"{mod}.{cls}.{name}") for name, mod, cls in started],
            [(res, name, sub) for res, name, mod, cls, sub in fails if sub])


freeze = dict(reversed(l.split()) for l in open(os.path.join(S, "ct11-freeze-01.txt"), encoding="utf-8") if l.strip())
text = open(os.path.join(S, "red-ct11-pre.txt"), encoding="utf-8").read()
rows, subs = rows_of(text)
out = ["CT-FIX-11 red first — this package's final test file in the CT worktree while scripts/prompt_gate.py was still",
       "0afba1bf5's (CT-FIX-10), before the code was written; run by the AI session. The test file was not changed after",
       "this run (its SHA-256 is the frozen one).", "",
       re.search(r"^Ran .*$", text, re.M).group(0) + " · " + re.search(r"^(OK|FAILED).*$", text, re.M).group(0), ""]
out += [f"  {res:5s} {tid}" for res, tid in rows]
out += ["", "failed subtests:"] + [f"  {res:5s} {name} {sub}" for res, name, sub in subs]
out += ["", "On the base source, with the queue unreadable: a governance record bound only by the queue row's record gets the",
        "false 'kuyruk satırının record sütunu bu kaydı göstermiyor' and no 'ölçülemedi' — with status approved (1) and with",
        "status draft, where the draft refusal itself is right (1). The other cases of the same test pass there: an explicit",
        "W1 (no unmeasured line), an explicit W10 and a broken field (their own refusals), and the two intact-queue controls.",
        "Controls.* pass."]
write("red-first.txt", out)

res = json.load(open(os.path.join(S, "guard/ct-guard-01.json"), encoding="utf-8"))
same = all(hashlib.sha256(open(os.path.join(S, "guard/ct-src", p), "rb").read()).hexdigest() == h for p, h in freeze.items())
out = ["CT-FIX-11 guard removal — one guard undone at a time in its own copy outside the repository; run by the AI session.",
       "Driver: drivers/harness14.py.txt (harness11 with its CT list replaced). The copy's two files equal the final",
       f"sources byte for byte: {same}. Each mutation: the file still parses; the named test fails; the control test",
       "(Controls.test_consumers_on_the_intact_fixture_exit_0) passes.", ""]
for r in res:
    out += [r["mutation"] + f"  ({r['file']})", f"    tests: {', '.join(t.split('.')[-1] for t in r['tests'])}",
            f"    applied: {r['applied']} · parses: {r['parsed']} · named: {'RED' if r['red'] else 'green'} — {r['summary']}"
            f" · control: {r['control_summary']} · proven: {r['proven']}"]
out += ["", f"proven: {sum(r['proven'] for r in res)} of {len(res)}"]
write("guard-removal.txt", out)

log = open(os.path.join(S, "ct11-run.log"), encoding="utf-8").read()
same_id = open(os.path.join(S, "ct11-pre.txt"), "rb").read() == open(os.path.join(S, "ct11-post.txt"), "rb").read()
out = ["CT-FIX-11 — the narrow component run agreed for this package, on the CT branch's two frozen files in the CT",
       "worktree (branch chore/ct/ct-fix-11 on 0afba1bf5 with this package's uncommitted changes). Not a full scripts/tests",
       "run, not a run on a combined candidate.", "",
       "commands (from the worktree root):",
       "  python3 -B scripts/ct_queue.py --self-test",
       "  python3 -B scripts/prompt_gate.py --self-test",
       "  python3 -B -m unittest -v scripts.tests.test_ct_cli_regression.CTFIX11_GovernanceUnknownRow"
       " scripts.tests.test_ct_cli_regression.CTFIX10_UnknownRowAndWriters"
       " scripts.tests.test_ct_cli_regression.CTFIX09_GateDiagnosisAndOneLine"
       " scripts.tests.test_ct_cli_regression.CTFIX04_I2_Binding"
       " scripts.tests.test_ct_cli_regression.CTFIX04_I3_PackIdentity"
       " scripts.tests.test_ct_cli_regression.Controls", "sources (SHA-256):"]
out += [f"  {p:50s}{h}" for p, h in freeze.items()]
out += ["result (from the full log):"] + ["  " + l for l in log.splitlines()
                                          if l.startswith(("### exit", "Ran ", "OK", "FAILED", "start ", "end ", "Python")) or "self-test:" in l]
out += [f"before = after (HEAD, branch, status, the two SHA-256): {same_id}",
        f"full log SHA-256 {sha('ct11-run.log')} (kept outside the repository)"]
write("checks.txt", out)
shutil.copyfile(os.path.join(S, "guard/harness14.py"), os.path.join(E, "drivers", "harness14.py.txt"))
shutil.copyfile(os.path.abspath(__file__), os.path.join(E, "drivers", "make_evidence15.py.txt"))
print("written", same, same_id)
