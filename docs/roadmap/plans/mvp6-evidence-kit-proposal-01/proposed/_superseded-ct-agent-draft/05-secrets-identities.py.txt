#!/usr/bin/env python3
"""CANDIDATE — NOT APPROVED.  Step 5: run-time secrets and fresh real-Auth identities.

generate : creates <W>/secrets/lane.env (mode 0600, umask 077) with fresh random values for JWT_SECRET,
           internal keys, module secret, every RANDOM_* placeholder named in <W>/env/*.env, and FIXTURE_PASSWORD.
           Nothing is printed.  The file lives only in the disposable workspace and is removed by step 9.
login    : for each identity in identities.tsv, performs a REAL Auth login through the lane Gateway
           (POST /api/tenant-auth/login with X-Tenant-Id — pattern from scenario_prepare.py) and records only
           status, presence booleans and a claim SUBSET (tenant_id, legal_entity_id present?, roles count).
           Tokens stay in process memory and are discarded; raw login/refresh bodies are never written.
seed     : GAP — no fixture-seeding script was archived by the accepted lanes (users/roles, Platform tenant record,
           MDM legal entities, position assignments).  Must be supplied per lane and reviewed; password hashes are
           BCrypt (Diten.AuthService.Infrastructure/Services/PasswordHasher.cs) and are secrets for evidence purposes.
"""
import argparse, base64, csv, json, os, re, secrets, stat, sys, urllib.error, urllib.request
from pathlib import Path

def generate(ws):
    sdir = Path(ws, "secrets"); os.umask(0o077); sdir.mkdir(mode=0o700, exist_ok=True)
    names = {"JWT_SECRET", "INTERNAL_EVENT_KEY", "PLATFORM_INTERNAL_KEY", "AUTH_INTERNAL_KEY", "MDM_MODULE_SECRET", "FIXTURE_PASSWORD"}
    for env in Path(ws, "env").glob("*.env"):
        names |= set(re.findall(r"@secret:([A-Z0-9_]+)", env.read_text()))
    f = sdir / "lane.env"
    if f.exists(): sys.exit("FAIL secrets already generated")
    with open(f, "w") as h:
        for n in sorted(names):
            v = secrets.token_urlsafe(48) if n != "FIXTURE_PASSWORD" else secrets.token_urlsafe(18) + "aA1!"
            h.write(f"{n}={v}\n")
    os.chmod(f, stat.S_IRUSR | stat.S_IWUSR)
    print(f"generated {len(names)} secrets (values not shown)")

def load_secrets(ws):
    return dict(l.split("=", 1) for l in Path(ws, "secrets/lane.env").read_text().splitlines() if "=" in l)

def claims(token):
    try:
        part = token.split(".")[1]; part += "=" * (-len(part) % 4)
        return json.loads(base64.urlsafe_b64decode(part))
    except Exception: return {}

def login(ws, gateway, identities, out):
    pw = load_secrets(ws)["FIXTURE_PASSWORD"]
    rows = []
    for ident in csv.DictReader(open(identities), delimiter="\t"):
        body = json.dumps({"email": ident["email"], "password": pw, "rememberMe": False}).encode()
        req = urllib.request.Request(gateway + "/api/tenant-auth/login", data=body, method="POST",
                                     headers={"Content-Type": "application/json", "X-Tenant-Id": ident["tenant_id"]})
        try:
            with urllib.request.urlopen(req, timeout=30) as r: status, payload = r.status, json.loads(r.read() or b"{}")
        except urllib.error.HTTPError as e: status, payload = e.code, {}
        data = (payload or {}).get("data") or {}
        c = claims(data.get("accessToken") or "")
        rows.append({"alias": ident["alias"], "status": status, "accessTokenPresent": bool(data.get("accessToken")),
                     "refreshTokenPresent": bool(data.get("refreshToken")),
                     "tenantMatches": str(c.get("tenant_id", "")).lower() == ident["tenant_id"].lower(),
                     "legalEntityPresent": "legal_entity_id" in c,
                     "legalEntityMatches": str(c.get("legal_entity_id", "")).lower() == (ident.get("legal_entity_id") or "").lower()})
        data.clear(); payload = None                          # drop token references immediately
    with open(Path(out, "LOGIN-CHECK.tsv"), "w", newline="") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0]), delimiter="\t", lineterminator="\n"); w.writeheader(); w.writerows(rows)
    bad = [r for r in rows if r["status"] != 200 or not r["accessTokenPresent"] or not r["tenantMatches"]]
    print("PASS" if not bad else "FAIL"); sys.exit(1 if bad else 0)

if __name__ == "__main__":
    a = argparse.ArgumentParser(); a.add_argument("mode", choices=["generate", "login"])
    a.add_argument("--workspace", required=True); a.add_argument("--gateway"); a.add_argument("--identities"); a.add_argument("--out")
    o = a.parse_args()
    generate(o.workspace) if o.mode == "generate" else login(o.workspace, o.gateway, o.identities, o.out)
