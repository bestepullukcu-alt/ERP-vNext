#!/usr/bin/env python3
"""CANDIDATE — NOT APPROVED.  Step 8: compare a before/after snapshot pair against the expected delta and list the
oplog write-set between them (replica set required — the reason step 3 always starts one).

expected.tsv columns: case, collection, delta (integer, or 0 for every negative case), state_changed (yes/no)
Usage: 08-db-diff.py --before b.json --after a.json --expected expected.tsv --case A08 --mongo-port P --db DB --out E
Output: <E>/DB-DELTA-<case>.tsv and <E>/DB-WRITESET-<case>.jsonl (namespace, op, _id only — no document bodies).
"""
import argparse, csv, json, subprocess, sys
from pathlib import Path

a = argparse.ArgumentParser()
for x in ("before", "after", "expected", "case", "db", "out"): a.add_argument("--" + x, required=True)
a.add_argument("--mongo-port", type=int, required=True)
o = a.parse_args()
if o.mongo_port == 27017: sys.exit("FAIL 27017 refused")
b, af = json.load(open(o.before)), json.load(open(o.after))
exp = {r["collection"]: r for r in csv.DictReader(open(o.expected), delimiter="\t") if r["case"] == o.case}
rows, fails = [], []
for c in sorted(set(b["collections"]) | set(af["collections"])):
    cb, ca = b["collections"].get(c, {"count": 0, "sha256": ""}), af["collections"].get(c, {"count": 0, "sha256": ""})
    delta, changed = ca["count"] - cb["count"], cb["sha256"] != ca["sha256"]
    e = exp.get(c)
    ok = e is not None and delta == int(e["delta"]) and changed == (e["state_changed"] == "yes")
    if not ok: fails.append(c)
    rows.append([o.case, c, cb["count"], ca["count"], delta, "yes" if changed else "no", e["delta"] if e else "MISSING", "PASS" if ok else "FAIL"])
js = ("const b=%s,a=%s;db.getSiblingDB('local').oplog.rs.find({ts:{$gt:Timestamp(b.t,b.i),$lte:Timestamp(a.t,a.i)},"
      "ns:{$regex:'^%s\\\\.'}}).forEach(e=>print(JSON.stringify({ns:e.ns,op:e.op,id:(e.o2&&e.o2._id)||(e.o&&e.o._id)||null})))") % (
      json.dumps(b["oplogTs"]["$timestamp"]), json.dumps(af["oplogTs"]["$timestamp"]), o.db)
ws = subprocess.run(["mongosh", "--quiet", "--host", "127.0.0.1", "--port", str(o.mongo_port), "--eval", js],
                    capture_output=True, text=True, check=True).stdout
Path(o.out, f"DB-WRITESET-{o.case}.jsonl").write_text(ws)
with open(Path(o.out, f"DB-DELTA-{o.case}.tsv"), "w", newline="") as f:
    w = csv.writer(f, delimiter="\t", lineterminator="\n")
    w.writerow(["case", "collection", "before", "after", "delta", "content_changed", "expected_delta", "result"]); w.writerows(rows)
    w.writerow([o.case, "*oplog-writes*", "", "", len([l for l in ws.splitlines() if l.strip()]), "", "", "INFO"])
print("PASS" if not fails else "FAIL " + ",".join(fails)); sys.exit(1 if fails else 0)
